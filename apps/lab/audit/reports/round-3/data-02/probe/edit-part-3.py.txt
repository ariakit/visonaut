def replace(old, new):
    global text
    assert text.count(old) == 1, (old[:70], text.count(old))

def between(start_mark, end_mark, new):
    global text
    start = text.index(start_mark)
    end = text.index(end_mark, start)
    text = text[:start] + new + text[end:]
