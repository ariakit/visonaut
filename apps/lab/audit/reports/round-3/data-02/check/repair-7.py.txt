# Repairs of the independent check, part 7: the last cuts. Each replacement
# must match exactly one time.

def swap(old, new, count=1):
    global text
    found = text.count(old)
    if found != count:
        raise SystemExit(f"expected {count} match, found {found}: {old[:90]!r}")
    text = text.replace(old, new)
