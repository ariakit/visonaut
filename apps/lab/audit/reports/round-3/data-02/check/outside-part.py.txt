# of the D-DATA-02 part and its decision panel.
import pathlib

    "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content/sections/40-data-storage.html"

end = '<div data-decision="D-DATA-02"></div>'

    part, tail = rest.split(end, 1)
    return head, part, tail
