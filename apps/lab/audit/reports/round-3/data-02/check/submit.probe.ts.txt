function megabytes(bytes: number) {
  return Math.round((bytes / 1_000_000) * 10) / 10;
}

async function live() {
  for (let pass = 0; pass < 4; pass++) {
    globalThis.gc?.();
    await wait(5);
  }
  const usage = process.memoryUsage();
  return usage.heapUsed + usage.arrayBuffers;
}

/** Live bytes that `make` adds while its result is held. */
async function held(make: () => unknown | Promise<unknown>) {
  const before = await live();
  let value: unknown = await make();
  const after = await live();
  expect(value).toBeTruthy();
  value = null;
  expect(value).toBe(null);
  return megabytes(after - before);
}

function slice(inventory: CaptureInventory, start: number, end: number): CaptureInventory {
  const manifest = inventory.manifest;
  const receipt = manifest?.localComparison;
  if (!manifest || !receipt) throw new Error("The probe needs a manifest with a receipt.");
  const captures = inventory.captures.slice(start, end);
  const records = new Map(inventory.profiles.map((record) => [record.digest, record]));
  const profiles = [...new Set(captures.map((capture) => capture.profileDigest))].map((digest) => {
    const record = records.get(digest);
    if (!record) throw new Error("The probe lost a profile.");
    return record;
  });
  const usedTests = new Set(captures.map((capture) => capture.testId));
  return {
    ...inventory,
    captures: captures.map((capture, index) => ({ ...capture, ordinal: index })),
    profiles,
    manifest: {
      ...manifest,
      profiles,
      tests: manifest.tests.filter((test) => usedTests.has(test.id)),
      captures: manifest.captures
        .slice(start, end)
        .map((capture, index) => ({ ...capture, ordinal: index })),
      localComparison: { ...receipt, captures: receipt.captures.slice(start, end) },
    },
  };
}

function pagesOf(inventory: CaptureInventory) {
  const pages: Buffer[] = [];
  const count = inventory.captures.length;
  for (let start = 0; start < count; start += pageSize) {
    const page = slice(inventory, start, Math.min(start + pageSize, count));
    pages.push(Buffer.from(canonicalJson(packInventory(page))));
  }
  return pages;
}

function listFromPages(pages: Buffer[]) {
  const captures: InventoryCapture[] = [];
  for (const page of pages) {
    const text = new TextDecoder("utf-8", { fatal: true }).decode(page);
    const parsed = JSON.parse(text) as PackedInventory;
    for (const capture of unpackInventory(parsed, { manifest: false }).captures) {
      captures.push(capture);
    }
  }
  return captures;
}

/** The complete manifest of today, built from pages of rows. */
function manifestFromPages(pages: Buffer[]): Manifest {
  let manifest: Manifest | undefined;
  for (const page of pages) {
    const text = new TextDecoder("utf-8", { fatal: true }).decode(page);
    const part = unpackInventory(JSON.parse(text) as PackedInventory).manifest;
    const receipt = part?.localComparison;
    if (!part || !receipt) throw new Error("The probe lost the manifest of a page.");
    if (!manifest?.localComparison) {
      manifest = part;
      continue;
    }
    manifest.profiles.push(...part.profiles);
    manifest.tests.push(...part.tests);
    manifest.captures.push(...part.captures);
    manifest.localComparison.captures.push(...receipt.captures);
  }
  if (!manifest) throw new Error("The probe made no manifest.");
  return manifest;
}

// The list of local-comparison.ts:186-208: a sorted copy, then one new object
// for each capture.
function referenceCaptures(baseline: InventoryCapture[]) {
  const captures = [...baseline].sort((first, second) => {
    const itemOrder = Buffer.compare(Buffer.from(first.itemKey), Buffer.from(second.itemKey));
    return (
      itemOrder || Buffer.compare(Buffer.from(first.variantKey), Buffer.from(second.variantKey))
    );
  });
  return captures.map((capture) => ({
    captureId: capture.id,
    itemKey: capture.itemKey,
    variantKey: capture.variantKey,
    profileDigest: capture.renderingProfileDigest,
    imageId: capture.image.id,
    image: {
      digest: capture.image.digest,
      mediaType: capture.image.contentType,
      bytes: capture.image.bytes,
      width: capture.image.width,
      height: capture.image.height,
    },
    path: `/v1/runs/run/reference/images/${capture.image.id}`,
  }));
}

// What validateLocalSubmission holds: the manifest, the reference list, 2 maps.
function validation(manifest: Manifest, baseline: InventoryCapture[]) {
  const receipt = manifest.localComparison;
  if (!receipt) throw new Error("The probe needs a receipt.");
  const captures = referenceCaptures(baseline);
  const originals = new Map(captures.map((capture) => [identityKey(capture), capture]));
  const results = new Map(receipt.captures.map((result) => [identityKey(result), result]));
  return { manifest, captures, originals, results };
}

// What materializeBundle holds before the write: the manifest, the reference
// inputs, 2 maps, and the new capture list (workflow-materialize.ts:275-386).
function materialization(manifest: Manifest, baseline: InventoryCapture[]) {
  const receipt = manifest.localComparison;
  if (!receipt) throw new Error("The probe needs a receipt.");
  const referenceInputs = baseline.map((capture) => ({
    id: capture.id,
    itemKey: capture.itemKey,
    variantKey: capture.variantKey,
    profileDigest: capture.profileDigest,
    renderingProfileDigest: capture.renderingProfileDigest,
    image: capture.image,
  }));
  const references = new Map(referenceInputs.map((capture) => [identityKey(capture), capture]));
  const results = new Map(receipt.captures.map((result) => [identityKey(result), result]));
  const tests = new Map(manifest.tests.map((test) => [test.id, test]));
  const captures = manifest.captures.map((capture) => {
    const key = identityKey({ itemKey: capture.itemKey, variantKey: capture.variant.key });
    const result = results.get(key);
    const representative = references.get(key)?.image;
    return {
      id: `run:${createHash("sha256").update(key).digest("hex")}`,
      itemKey: capture.itemKey,
      variantKey: capture.variant.key,
      ordinal: capture.ordinal,
      imageId: representative?.id,
      image: representative,
      profileDigest: capture.profileDigest,
      renderingProfileDigest: createHash("sha256").update(`r${capture.profileDigest}`).digest("hex"),
      environmentProfileDigest: createHash("sha256")
        .update(`e${capture.profileDigest}`)
        .digest("hex"),
      testId: capture.testId,
      testRetry: capture.testRetry,
      metadata: {
        name: capture.name ?? capture.itemKey,
        variant: capture.variant,
        profile: { $visonautProfileDigest: capture.profileDigest },
        source: tests.get(capture.testId),
        localMode: receipt.mode,
        observedImage: capture.image,
        candidateStored: result?.outcome !== "unchanged",
        comparison: capture.comparison,
        comparisonDigest: createHash("sha256").update("c").digest("hex"),
        localResult: {
          outcome: result?.outcome,
          changedPixels: result?.changedPixels,
          ratio: result?.ratio,
          engineVersion: receipt.engineVersion,
          codecVersion: receipt.codecVersion,
          maskExpected: !!result?.mask,
        },
      },
    };
  });
  return { manifest, referenceInputs, references, results, captures };
}

it("measures Submit validation and materialization with complete lists from pages", async () => {
  const stored = readFileSync(`${data}/run-inventory.json`);
  const document = JSON.parse(stored.toString("utf8"));
  const digest = createHash("sha256").update(stored).digest("hex");
  const facts = await readCaptureInventory(
    {
      async get() {
        return {
          size: stored.byteLength,
          body: new ReadableStream<Uint8Array>({
            start(controller) {
              controller.enqueue(stored.slice());
              controller.close();
            },
          }),
        };
      },
    },
    {
      objectKey: `runs/${document.runId}/inventory/${digest}.json`,
      digest,
      bytes: stored.byteLength,
      captureCount: document.captures.length,
    },
  );
  const rows = [];
  for (const count of [3832, 10000, 20000, 40000]) {
    const scaled = count === facts.captures.length ? facts : scaleInventory(facts, count);
    const pages = pagesOf(scaled);
    const check = manifestFromPages(pages);
    expect(check.captures.length).toBe(count);
    expect(check.localComparison?.captures.length).toBe(count);
    rows.push({
      captures: count,
      profiles: check.profiles.length,
      manifestMB: await held(() => manifestFromPages(pages)),
      baselineListMB: await held(() => listFromPages(pages)),
      // Held after the baseline list is released.
      validationMB: await held(() => validation(manifestFromPages(pages), listFromPages(pages))),
      // The largest point of validation: the baseline list is still alive.
      validationWithBaselineMB: await held(() => {
        const baseline = listFromPages(pages);
        return { baseline, value: validation(manifestFromPages(pages), baseline) };
      }),
      materializationMB: await held(() =>
        materialization(manifestFromPages(pages), listFromPages(pages)),
      ),
    });
  }
  const result = {
    runtime: process.version,
    note: "Live V8 heap plus array buffers that the named value adds, after 4 collections. Node, not workerd. The canonical JSON texts, the digests, and the D1 statements of the real steps are not in these numbers.",
    pageSize,
    rows,
  };
  writeFileSync(`${lane}/check/submit.json`, `${JSON.stringify(result, null, 2)}\n`);
  console.table(rows);
});
