const read = (dir, file) => readFileSync(join(dir, file), "utf8");
const sha = (text) => createHash("sha256").update(text).digest("hex").slice(0, 16);
let failures = 0;
function report(ok, text) {
  if (!ok) failures += 1;
  console.log(`${ok ? "PASS" : "FAIL"} ${text}`);
}

// findings.json
const findingsNow = read(now, "findings.json");
const findingsOld = read(old, "findings.json");

// record.json
const recordNow = JSON.parse(read(now, "record.json"));
const recordOld = JSON.parse(read(old, "record.json"));
const recordDiff = [];
function walk(a, b, path) {
  if (typeof a !== typeof b || Array.isArray(a) !== Array.isArray(b)) {
    recordDiff.push(path);
    return;
  }
  if (a && typeof a === "object") {
    const keys = new Set([...Object.keys(a), ...Object.keys(b)]);
    for (const key of keys) {
      if (!(key in a) || !(key in b)) recordDiff.push(`${path}.${key}`);
      else walk(a[key], b[key], `${path}.${key}`);
    }
    return;
  }
  if (a !== b) recordDiff.push(path);
}
walk(recordNow, recordOld, "record");
report(
  recordDiff.length === 2 && recordDiff.includes("record.revision") && recordDiff.includes("record.date"),
  `record.json differs only in: ${recordDiff.join(", ")}`,
);

// Field level compare with its own collector.
function fieldDiff(a, b, path, out) {
  if (typeof a !== typeof b || Array.isArray(a) !== Array.isArray(b)) return void out.push(path);
  if (a && typeof a === "object") {
    const keys = new Set([...Object.keys(a), ...Object.keys(b)]);
    for (const key of keys) {
      if (!(key in a) || !(key in b)) out.push(`${path}.${key}`);
      else fieldDiff(a[key], b[key], `${path}.${key}`, out);
    }
    return;
  }
  if (a !== b) out.push(path);
}
const allowed = new Set(["D-PRE-08", "D-PRE-09", "D-PRE-10", "D-PRE-11", "D-PRE-12"]);
for (let i = 0; i < decisionsNow.length; i += 1) {
  const out = [];
  fieldDiff(decisionsNow[i], decisionsOld[i], decisionsNow[i].id, out);
  for (const item of out) changed.push(item);
}
console.log(`     changed fields: ${changed.join(", ")}`);
report(

);
for (let i = 0; i < decisionsNow.length; i += 1) {

}

// Placeholders.
