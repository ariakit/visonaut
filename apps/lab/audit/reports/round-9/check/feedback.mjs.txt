let failures = 0;
const lines = [];
function report(ok, text) {
  if (!ok) failures += 1;
  lines.push(`${ok ? "PASS" : "FAIL"} ${text}`);
  console.log(`${ok ? "PASS" : "FAIL"} ${text}`);
}

const browser = await chromium.launch({ channel: "chrome" });
const requests = [];
async function open(viewport) {
  const context = await browser.newContext({ viewport, deviceScaleFactor: 1 });
  await context.route("**/*", (route) => {
    const request = route.request();
    const host = new URL(request.url()).host;
    requests.push(`${request.method()} ${host}`);
    if (request.method() !== "GET" || /visonaut\.com$/.test(host)) return route.abort();
    return route.continue();
  });
  const page = await context.newPage();
  const errors = [];
  page.on("pageerror", (error) => errors.push(String(error)));
  page.on("console", (message) => {
    if (message.type() === "error") errors.push(message.text());
  });
  await page.goto(url, { waitUntil: "load" });
  await page.locator("#doc[data-ready]").waitFor({ timeout: 15000 });
  return { context, page, errors };
}
const count = async (page) => (await page.locator("#answered-count").innerText()).trim();
const prompt = (page) => page.locator("#prompt-text").inputValue();
async function ready(page) {
  await page.reload({ waitUntil: "load" });
  await page.locator("#doc[data-ready]").waitFor({ timeout: 15000 });
}

// 8. Layout at two widths, with pictures.
for (const [name, viewport] of [
  ["1440", { width: 1440, height: 1000 }],
  ["400", { width: 400, height: 900 }],
]) {
  const { context, page, errors } = await open(viewport);
  const facts = await page.evaluate(() => {
    const anchor = document.getElementById("your-answers-plan-table");
    let node = anchor.nextElementSibling;
    while (node && !node.querySelector?.("table") && node.tagName !== "TABLE") node = node.nextElementSibling;
    const table = node.tagName === "TABLE" ? node : node.querySelector("table");
    const box = table.parentElement;
    const rows = [...table.querySelectorAll("tbody tr")];
    const links = rows.map((row) => {
      const link = row.querySelector('a[href^="https://github.com/ariakit/visonaut/issues/"]');
      return { no: Number(row.querySelector("td.num").textContent), href: link?.getAttribute("href"), text: link?.textContent.trim(), key: row.querySelector("th code")?.textContent, target: link?.target, rel: link?.rel };
    });
    const wide = [...document.querySelectorAll("main *")]
      .filter((el) => el.getBoundingClientRect().right > document.documentElement.clientWidth + 1 && !el.closest(".table-wrap, .table-scroll, pre, [data-demo], .demo, table, .code, .findings, #findings-explorer"))
      .slice(0, 5)
      .map((el) => `${el.tagName}.${el.className}#${el.id}`);
    return {
      pageScroll: document.documentElement.scrollWidth - document.documentElement.clientWidth,
      bodyScroll: document.body.scrollWidth - document.documentElement.clientWidth,
      tableWidth: Math.round(table.getBoundingClientRect().width),
      boxClient: box.clientWidth,
      boxScroll: box.scrollWidth,
      boxClass: box.className,
      boxOverflowX: getComputedStyle(box).overflowX,
      headers: [...table.querySelectorAll("thead th")].map((cell) => cell.textContent.trim()),
      links,
      wide,
    };
  });
  report(facts.pageScroll === 0 && facts.bodyScroll <= 0, `${name} px: the page does not scroll sideways (document ${facts.pageScroll} px, body ${facts.bodyScroll} px)`);
  report(facts.links.length === 22 && facts.links.every((l) => l.href === `https://github.com/ariakit/visonaut/issues/${254 + l.no}` && l.text === `#${254 + l.no}`), `${name} px: the plan table has 22 rows, and each issue link is 254 plus the plan number`);
  report(facts.links.every((l) => l.target === "_blank"), `${name} px: each issue link opens in a new tab`);
  console.log(`     ${name} px: table ${facts.tableWidth} px in a box of ${facts.boxClient} px (scroll width ${facts.boxScroll} px, overflow-x ${facts.boxOverflowX}, class "${facts.boxClass}"), headers: ${facts.headers.join(" | ")}`);
  if (name === "1440") report(facts.boxScroll <= facts.boxClient, "1440 px: the plan table fits its box");
  else report(facts.boxScroll > facts.boxClient && /auto|scroll/.test(facts.boxOverflowX), "400 px: the plan table scrolls in its own box");
  console.log(`     ${name} px: elements outside the viewport and outside a scroll box: ${facts.wide.join(", ") || "none"}`);
  // Scroll the whole page and measure again: a sideways scroll can come from content lower in the page.
  const maxScrollX = await page.evaluate(async () => {
    let max = 0;
    for (let y = 0; y < document.documentElement.scrollHeight; y += 1500) {
      window.scrollTo(0, y);
      await new Promise((resolve) => requestAnimationFrame(resolve));
      window.scrollBy(50, 0);
      max = Math.max(max, window.scrollX);
      window.scrollTo(0, y);
    }
    return max;
  });
  report(maxScrollX === 0, `${name} px: a sideways scroll of the window moves 0 px at each place of the page (${maxScrollX})`);
  for (const [label, anchor] of [
    ["summary", "summary"],
    ["summary-start", "summary-start"],
    ["section-15-top", "your-answers"],
    ["done", "your-answers-done"],
    ["left", "your-answers-left"],
    ["round-6-table", "your-answers-round-6"],
    ["plan-table", "your-answers-plan-table"],
  ]) {

      document.getElementById(id).scrollIntoView({ block: "start" });
      window.scrollBy(0, -70);
    }, anchor);
    await page.waitForTimeout(350);
    await page.screenshot({ path: `${out}/${label}-${name}.png` });
  }
  // The sticky bar stays visible at the plan table.
  report(await page.locator("#copy-prompt-button").isVisible(), `${name} px: the sticky bar is visible at the plan table`);
  report(errors.length === 0, `${name} px: no console error (${errors.slice(0, 3).join(" | ")})`);
  await context.close();
