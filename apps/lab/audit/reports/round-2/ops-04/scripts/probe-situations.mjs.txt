//   A. unchanged files: accepted. Edited app.yml: refused at rest.direct_workflow_blob.
//      Edited ci.yml: refused at rest.caller_workflow_blob.
//   B. accepted with 2 GitHub requests fewer (no "contents" read), for any blob of the

//      and another commit are refused at claim.job_workflow_ref or claim.job_workflow_sha.

  resolve(specifier, context, nextResolve) {
    try {
      return nextResolve(specifier, context);
    } catch (error) {
      if (specifier.startsWith(".") && specifier.endsWith(".js")) {
        return nextResolve(`${specifier.slice(0, -3)}.ts`, context);
      }
      throw error;
    }
  },
});

function sign(claims) {
  const now = Math.floor(Date.now() / 1000);
  return new jose.SignJWT(claims)
    .setProtectedHeader({ alg: "RS256", kid: "local" })
    .setIssuer("https://token.actions.githubusercontent.com")
    .setAudience(`${origin}/submit`)
    .setJti(crypto.randomUUID())
    .setIssuedAt(now)
    .setNotBefore(now)
    .setExpirationTime(now + 300)
    .sign(privateKey);
}

function githubStub({ callerBlob = blobC, appBlob = blobB, submitJobName, calls }) {
  return {
    appId: "5028451",
    repository,
    repositoryId: "104133653",
    async request(path) {
      calls.push(path.replace(`/repos/${repository}`, ""));
      if (path.includes("/contents/.github/workflows/ci.yml")) {
        return { type: "file", path: ".github/workflows/ci.yml", sha: callerBlob };
      }
      if (path.includes("/contents/.github/workflows/app.yml")) {
        return { type: "file", path: ".github/workflows/app.yml", sha: appBlob };
      }
      if (path.endsWith("/actions/runs/9001")) {
        return { id: 9001, run_attempt: 1, status: "in_progress", conclusion: null };
      }
      if (path.endsWith("/actions/runs/9001/attempts/1")) {
        return {
          id: 9001,
          run_attempt: 1,
          repository: { id: 104133653, owner: { id: 40200111 } },
          event: "pull_request",
          path: ".github/workflows/ci.yml",
          status: "in_progress",
          conclusion: null,
          head_sha: sourceHead,
          head_branch: "feature",
        };
      }
      if (path.includes("/attempts/1/jobs")) {
        const job = (id, name, check) => ({
          id,
          name,
          run_id: 9001,
          run_attempt: 1,
          status: "in_progress",
          conclusion: null,
          check_run_url: `https://api.github.com/repos/${repository}/check-runs/${check}`,
        });
        return {
          jobs: [
            job(7001, submitJobName, 555),
            job(7002, "Plan", 556),
            job(7003, "App / Visual Capture (linux)", 557),
          ],
        };
      }
      if (path.endsWith("/pulls/7")) {
        return {
          state: "open",
          merge_commit_sha: testedSha,
          base: { ref: "main", repo: { id: 104133653 } },
          head: { ref: "feature", sha: sourceHead, repo: { id: 104133653 } },
        };
      }
      if (path.endsWith("/git/ref/pull/7/merge")) return { object: { sha: testedSha } };
      if (path.endsWith("/git/ref/heads/main")) return { object: { sha: mainSha } };
      if (path.endsWith(`/git/commits/${testedSha}`)) {
        return { parents: [{ sha: mainSha }, { sha: sourceHead }] };
      }
      throw new Error(`unexpected GitHub path ${path}`);
    },
  };
}

const warnings = [];
console.warn = (line) => warnings.push(String(line));

async function verify({ label, claims, configuration, stub = {} }) {
  warnings.length = 0;
  const calls = [];
  const submitJobName = configuration.shards[0]?.jobName ?? "";
  let result;
  try {
    await security.verifyGitHubOidc({
      token: await sign(claims),
      request,
      configuration,
      github: githubStub({ ...stub, submitJobName, calls }),
      keySet,
    });
    result = { accepted: true };
  } catch (error) {
    const check = warnings.map((line) => JSON.parse(line).check)[0];
    result = { accepted: false, code: error?.code, status: error?.status, check };
  }
  const contentReads = calls.filter((path) => path.includes("/contents/")).length;
  console.log(
    JSON.stringify({ case: label, ...result, githubRequests: calls.length, contentReads }),
  );
}
