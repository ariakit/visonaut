// Experiment 3 (checker). Question: for a job two levels below the top workflow
// (caller -> reusable workflow -> reusable workflow), which file does GitHub name in
// the job_workflow_ref claim, and which form does a tag reference have?
//
// Data: one public provenance file of slsa-verifier (its own test data). The signing
// job runs in a "delegator" workflow that a "builder" workflow calls, and the user
// workflow calls the builder. Fulcio copies the claims of the GitHub OIDC token into
// the certificate: extension 1.3.6.1.4.1.57264.1.9 is job_workflow_ref, and
// 1.3.6.1.4.1.57264.1.18 is workflow_ref.
//
// Expected before the run: extension 1.9 names the delegator file (the inner file)
// with "@refs/tags/<tag>", and extension 1.18 names the top workflow of the user.
//   node probe-nested-claim.mjs
import { X509Certificate } from "node:crypto";
import { readFile } from "node:fs/promises";
import { dirname, join } from "node:path";
import { fileURLToPath } from "node:url";

const lane = join(dirname(fileURLToPath(import.meta.url)), "..");
const file = join(lane, "evidence", "nested", "binary-linux-amd64-push-v14.build.slsa");

const extensionNames = {
  "1.3.6.1.4.1.57264.1.9": "job_workflow_ref (Build Signer URI)",
  "1.3.6.1.4.1.57264.1.10": "job_workflow_sha (Build Signer Digest)",
  "1.3.6.1.4.1.57264.1.12": "repository (Source Repository URI)",
  "1.3.6.1.4.1.57264.1.18": "workflow_ref (Build Config URI)",
  "1.3.6.1.4.1.57264.1.19": "workflow_sha (Build Config Digest)",
};

function findCertificates(value, found) {
  if (typeof value === "string") {
    if (value.startsWith("-----BEGIN CERTIFICATE")) found.push(value);
    // A bundle holds the certificate as base64 of the DER bytes or of the PEM text.
    if (/^[A-Za-z0-9+/=]{800,}$/.test(value)) {
      const bytes = Buffer.from(value, "base64");
      const text = bytes.toString("utf8");
      if (text.startsWith("-----BEGIN CERTIFICATE")) found.push(text);
      else found.push(bytes);
    }
    return;
  }
  if (!value || typeof value !== "object") return;
  for (const entry of Object.values(value)) {
    findCertificates(entry, found);
  }
}

// Reads the UTF8String values of the named extensions from the DER bytes. The
// extension value follows the object identifier as OCTET STRING { UTF8String }.
function readExtensions(der) {
  const result = {};
  for (const [oid, name] of Object.entries(extensionNames)) {
    const parts = oid.split(".").map(Number);
    const encoded = [parts[0] * 40 + parts[1]];
    for (const part of parts.slice(2)) {
      const groups = [part & 0x7f];
      let rest = part >> 7;
      while (rest > 0) {
        groups.unshift((rest & 0x7f) | 0x80);
        rest >>= 7;
      }
      encoded.push(...groups);
    }
    const needle = Buffer.from([0x06, encoded.length, ...encoded]);
    const start = der.indexOf(needle);
    if (start < 0) continue;
    // After the identifier: OCTET STRING (tag 0x04), then UTF8String (tag 0x0c).
    let offset = start + needle.length;
    const readLength = () => {
      const first = der[offset++];
      if (first < 0x80) return first;
      let length = 0;
      for (let i = 0; i < (first & 0x7f); i++) {
        length = (length << 8) | der[offset++];
      }
      return length;
    };
    if (der[offset++] !== 0x04) continue;
    readLength();
    if (der[offset++] !== 0x0c) continue;
    const length = readLength();
    result[name] = der.subarray(offset, offset + length).toString("utf8");
  }
  return result;
}

const lines = (await readFile(file, "utf8")).split("\n").filter(Boolean);
for (const line of lines) {
  const found = [];
  const bundle = JSON.parse(line);
  findCertificates(bundle, found);
  for (const candidate of found) {
    let certificate;
    try {
      certificate = new X509Certificate(candidate);
    } catch {
      continue;
    }
    console.log(JSON.stringify({ issued: certificate.validFrom }));
    console.log(JSON.stringify(readExtensions(certificate.raw), null, 2));
  }
  // The signed statement names the workflow that the user workflow called.
  const payload = bundle.dsseEnvelope?.payload ?? bundle.payload;
  if (payload) {
    const statement = JSON.parse(Buffer.from(payload, "base64").toString("utf8"));
    const predicate = statement.predicate ?? {};
    console.log(
      JSON.stringify(
        {
          builderId: predicate.runDetails?.builder?.id ?? predicate.builder?.id,
          buildType: predicate.buildDefinition?.buildType ?? predicate.buildType,
        },
        null,
        2,
      ),
    );
  }
}
