// Applies decisions.patch.json and terms.patch.json of this lane to the copy of
// the record content in ./content. It changes no file of the repository.
const fs = require("node:fs");
const path = require("node:path");

const lane = __dirname;

function readJson(file) {
  return JSON.parse(fs.readFileSync(path.join(lane, file), "utf8"));
}

function writeJson(file, value) {
  fs.writeFileSync(path.join(lane, file), `${JSON.stringify(value, null, 2)}\n`);
}

const decisions = readJson("content/decisions.json");
for (const patch of readJson("decisions.patch.json")) {
  const index = decisions.findIndex((decision) => decision.id === patch.id);
  if (index < 0) {
    throw new Error(`Unknown decision: ${patch.id}`);
  }
  const recommended = patch.options.filter((option) => option.recommended).length;
  if (recommended > 1) {
    throw new Error(`${patch.id} has more than one recommended option.`);
  }
  decisions[index] = patch;
}
writeJson("content/decisions.json", decisions);

const terms = readJson("content/terms.json");
for (const patch of readJson("terms.patch.json")) {
  const index = terms.findIndex((term) => term.term === patch.term);
  if (index < 0) {
    terms.push(patch);
  } else {
    terms[index] = patch;
  }
}
writeJson("content/terms.json", terms);

console.log("applied");
