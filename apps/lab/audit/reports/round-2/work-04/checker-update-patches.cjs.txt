// Checker repairs of decisions.patch.json and terms.patch.json of this lane.
// It reads the record of the repository and writes only in the lane folder.
const fs = require("node:fs");
const path = require("node:path");

const lane = __dirname;
const record =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content";

function readJson(file) {
  return JSON.parse(fs.readFileSync(file, "utf8"));
}

function option(decision, id) {
  const found = decision.options.find((entry) => entry.id === id);
  if (!found) {
    throw new Error(`Missing option ${decision.id}/${id}`);
  }
  return found;
}

const patch = readJson(path.join(lane, "decisions.patch.json"));
const base = readJson(path.join(record, "decisions.json"));
