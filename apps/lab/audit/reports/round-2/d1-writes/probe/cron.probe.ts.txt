/** Stubs the GitHub App delivery API. Every listed delivery is a successful delivery. */
function stubGitHub(deliveries: number, origin: string) {
  const entries: string[] = [];
  for (let index = 0; index < deliveries; index += 1) {
    const guid = `00000000-0000-4000-8000-${String(index).padStart(12, "0")}`;
    entries.push(
      `{"id":${1000 + index},"guid":"${guid}","status_code":200,"delivered_at":"2026-10-05T10:00:00Z","event":"push","repository_id":123,"installation_id":34}`,
    );
  }
  return async (input: URL | string | Request) => {
    const url = new URL(String(input instanceof Request ? input.url : input));
    if (url.pathname === "/app/hook/config") {
      return new Response(JSON.stringify({ url: `${origin}/v1/webhooks` }));
    }
    if (url.pathname === "/app/hook/deliveries") return new Response(`[${entries.join(",")}]`);
    return new Response("{}", { status: 404 });
  };
}
