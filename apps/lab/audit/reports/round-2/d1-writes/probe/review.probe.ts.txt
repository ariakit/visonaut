interface Call {
  kind: "single" | "batch";
  start: number;
  end: number;
  statements: {
    sql: string;
    values: unknown[];
    rows_read: number;
    rows_written: number;
    duration: number;
    rows: number;
  }[];
}

function measure(native: any) {
  let calls: Call[] = [];
  let enabled = false;
  const info = new WeakMap<object, { native: any; sql: string; values: unknown[] }>();
  const single = async (stmt: any, sql: string, values: unknown[], method: "all" | "run") => {
    const start = performance.now();
    const result = await stmt[method]();
    const end = performance.now();
    if (enabled)
      calls.push({
        kind: "single",
        start,
        end,
        statements: [
          {
            sql,
            values,
            rows_read: result.meta?.rows_read ?? 0,
            rows_written: result.meta?.rows_written ?? 0,
            duration: result.meta?.duration ?? 0,
            rows: result.results?.length ?? 0,
          },
        ],
      });
    return result;
  };
  const wrap = (stmt: any, sql: string, values: unknown[]): any => {
    const wrapped = {
      bind: (...next: unknown[]) => wrap(stmt.bind(...next), sql, next),
      async first(column?: string) {
        const result = await single(stmt, sql, values, "all");
        const row = result.results?.[0] ?? null;
        return column === undefined ? row : (row?.[column] ?? null);
      },
      all: () => single(stmt, sql, values, "all"),
      run: () => single(stmt, sql, values, "run"),
      raw: (...args: unknown[]) => stmt.raw(...args),
    };
    info.set(wrapped, { native: stmt, sql, values });
    return wrapped;
  };
  const database = {
    prepare: (sql: string) => wrap(native.prepare(sql), sql, []),
    async batch(list: object[]) {
      const entries = list.map((entry) => {
        const found = info.get(entry);
        if (!found) throw new Error("Expected a measured statement.");
        return found;
      });
      const start = performance.now();
      const results = await native.batch(entries.map((entry) => entry.native));
      const end = performance.now();
      if (enabled)
        calls.push({
          kind: "batch",
          start,
          end,
          statements: entries.map((entry, index) => ({
            sql: entry.sql,
            values: entry.values,
            rows_read: results[index]?.meta?.rows_read ?? 0,
            rows_written: results[index]?.meta?.rows_written ?? 0,
            duration: results[index]?.meta?.duration ?? 0,
            rows: results[index]?.results?.length ?? 0,
          })),
        });
      return results;
    },
    exec: (sql: string) => native.exec(sql),
    dump: () => native.dump(),
  };
  return {
    database,
    start() {
      calls = [];
      enabled = true;
    },
    stop() {
      enabled = false;
      return calls;
    },
    peek() {
      return calls;
    },
  };
}

function summarize(name: string, calls: Call[], wallMs: number, note?: string) {
  const statements = calls.flatMap((call) => call.statements);
  // Longest chain of calls that do not overlap in time: a lower bound on sequential waits.
  const sorted = [...calls].sort((a, b) => a.end - b.end);
  let depth = 0;
  let cursor = -Infinity;
  for (const call of sorted) {
    if (call.start >= cursor) {
      depth += 1;
      cursor = call.end;
    }
  }
  return {
    name,
    note,
    d1Calls: calls.length,
    batches: calls.filter((call) => call.kind === "batch").length,
    statements: statements.length,
    sequentialDepth: depth,
    rowsRead: statements.reduce((sum, entry) => sum + entry.rows_read, 0),
    rowsWritten: statements.reduce((sum, entry) => sum + entry.rows_written, 0),
    sqlDurationMs: Number(statements.reduce((sum, entry) => sum + entry.duration, 0).toFixed(2)),
    localWallMs: Number(wallMs.toFixed(1)),
    bySql: [
      ...statements.reduce((map, entry) => {
        const key = entry.sql.replace(/\s+/gu, " ").slice(0, 200);
        const current = map.get(key) ?? { count: 0, rows_read: 0, rows_written: 0 };
        current.count += 1;
        current.rows_read += entry.rows_read;
        current.rows_written += entry.rows_written;
        map.set(key, current);
        return map;
      }, new Map<string, { count: number; rows_read: number; rows_written: number }>()),
    ]
      .map(([sql, value]) => ({ ...value, sql }))
      .sort((a, b) => b.rows_read - a.rows_read || b.count - a.count)
      .slice(0, 100000),
    top: [...statements]
      .sort((a, b) => b.rows_read - a.rows_read)
      .slice(0, 8)
      .map((entry) => ({
        rows_read: entry.rows_read,
        rows_written: entry.rows_written,
        rows: entry.rows,
        durationMs: Number(entry.duration.toFixed(2)),
        sql: entry.sql.replace(/\s+/gu, " ").slice(0, 260),
      })),
    order: calls.map((call) => ({
      kind: call.kind,
      startMs: Number((call.start - (calls[0]?.start ?? 0)).toFixed(1)),
      endMs: Number((call.end - (calls[0]?.start ?? 0)).toFixed(1)),
      rows_read: call.statements.reduce((sum, entry) => sum + entry.rows_read, 0),
      sql: call.statements.map((entry) => entry.sql.replace(/\s+/gu, " ").slice(0, 110)),
    })),
  };
}

const tableCounts = async (raw: any) => {
  const tables = await raw
    .prepare(
      "SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' AND name NOT LIKE '_cf_%' ORDER BY name",
    )
    .all();
