/** Changes the size of the screenshot list before the measurement. */
async function prepare(page, frame, prep) {
  if (!prep || prep === "none") return;
  const combo = frame.getByRole("combobox", { name: /^Status:/ }).locator("visible=true").first();
  await combo.click();
  await frame.getByRole("option", { name: /^Unchanged/ }).first().click();
  await settle(page, 500);
  if (prep === "unchanged") return;
  const list = frame.getByRole("navigation", { name: "Screenshots" });
  if (prep === "button") {
    await list.getByRole("button", { name: /^button/ }).first().click();
    await settle(page, 500);
    return;
  }
  // Open every folder. A click can move the others, so look again each time.
  for (let guard = 0; guard < 60; guard += 1) {
    const closed = list.locator('button[aria-expanded="false"]');
    if (!(await closed.count())) break;
    await closed.first().click();
    await page.waitForTimeout(60);
  }
  await settle(page, 800);
}""",
)
