// Saves one style sheet as a browser gets it for a link.
// Usage: node fetch-css.mjs <url> <out file>
import { writeFileSync } from "node:fs";

const [url, out] = process.argv.slice(2);
const started = performance.now();
const response = await fetch(url, { headers: { accept: "text/css,*/*;q=0.1" } });
const text = await response.text();
writeFileSync(out, text);
console.log(
  `${response.status} ${Buffer.byteLength(text)} bytes in ${Math.round(performance.now() - started)} ms -> ${out}`,
);
