// Prints the short form of a result file of attribute.mjs or growth.mjs.
// Usage: node show.mjs <attribute|growth> <json>
import { readFileSync } from "node:fs";

const [kind, file] = process.argv.slice(2);
const data = JSON.parse(readFileSync(file, "utf8"));
if (kind === "attribute") {
  console.log(data.load);
  console.log(JSON.stringify({ files: data.scannedFiles, candidates: data.candidates, validClasses: data.validClasses, build: data.build, imports: data.imports }, null, 1));
  console.table(
    Object.entries(data.groups).map(([name, group]) => ({
      group: name.slice(0, 58),
      files: group.files,
      candidates: group.candidates,
      valid: group.validClasses,
      validOnlyHere: group.validClassesOnlyHere,
      aloneRaw: group.aloneAddsToEmpty.raw,
      aloneGzip: group.aloneAddsToEmpty.gzip,
      savesRaw: group.removalSaves.raw,
      savesGzip: group.removalSaves.gzip,
    })),
  );
  console.table(data.cumulative.map((step) => ({ ...step, step: step.step.slice(0, 58) })));
  console.table(
    Object.entries({ ...data.byUtility, ...data.byVariant }).map(([name, value]) => ({
      split: name,
      classes: value.classes,
      aloneRaw: value.aloneAddsToEmpty.raw,
      aloneGzip: value.aloneAddsToEmpty.gzip,
      savesRaw: value.removalSaves.raw,
      savesGzip: value.removalSaves.gzip,
    })),
  );
} else {
  console.log(data.load);
  console.table(data.steps.map(({ step, ...rest }) => ({ step: step.slice(0, 64), ...rest })));
  console.log(JSON.stringify(data.primitivesUsedByTheLabSource, null, 1));
  if (data.design) console.log(JSON.stringify(data.design, null, 1));
}
