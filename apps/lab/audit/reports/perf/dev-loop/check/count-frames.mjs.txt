// Checker: how many preview frames does a lab page load at a window size, and
// how many of them are in the viewport (what `rootMargin: "0px"` would load)?
// The documents of the frames are aborted, so the page stays light.
// Usage: node count-frames.mjs <origin> <path> <width> <height> [...more heights]
import { createRequire } from "node:module";

const root = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const require = createRequire(`${root}/package.json`);
const { chromium } = require("@playwright/test");

const [origin, path, widthText, ...heights] = process.argv.slice(2);
const browser = await chromium.launch({ channel: "chrome" });
for (const heightText of heights) {
  const width = Number(widthText);
  const height = Number(heightText);
  const context = await browser.newContext({ viewport: { width, height } });
  await context.route(
    (url) => url.pathname.startsWith("/preview/"),
    (route) => route.fulfill({ status: 200, contentType: "text/html", body: "<!doctype html><title>x</title>" }),
  );
  const page = await context.newPage();
  await page.goto(`${origin}${path}`, { waitUntil: "load" });
  await page.waitForTimeout(2500);
  const result = await page.evaluate(() => {
    const frames = [...document.querySelectorAll("iframe")];
    const inView = frames.filter((frame) => {
      const box = frame.parentElement.getBoundingClientRect();
      return box.bottom > 0 && box.top < innerHeight && box.right > 0 && box.left < innerWidth;
    });
    const sizes = frames.map((frame) => `${frame.clientWidth}x${frame.clientHeight}`);
    return {
      frames: frames.length,
      inViewport: inView.length,
      pageHeight: document.documentElement.scrollHeight,
      frameSizes: [...new Set(sizes)],
    };
  });
  console.log(`${path} at ${width}x${height}: ${JSON.stringify(result)}`);
  await context.close();
}
await browser.close();
