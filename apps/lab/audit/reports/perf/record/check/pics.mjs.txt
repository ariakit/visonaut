// Checker: pictures of the list in each state, for each form, compared with
// the first form pixel by pixel.
//
// Usage: node pics.mjs --name <result> --view large|all --theme dark|light \
//   --conds base:4365,bar:4366,fixsib:4365:lab-fixsib
//
// States: selected (a row is selected, the focus is not in the list),
// focus-row (keyboard focus on the selected row), hover-row (the pointer on a
// row that is not selected), hover-selected, hover-gap (the pointer between
// two rows, after 700 ms), hover-out (the pointer leaves the list), and in
// the view "all": hover-button and focus-button (a folder button).
// A pixel differs when one channel differs by more than 2.
import { appendFileSync, writeFileSync } from "node:fs";
import { args, check, chromium, load, open, quiet, saveResult, sheetOf } from "./clib.mjs";

const options = args();
const name = options.name ?? "pics";
const view = options.view ?? "large";
const theme = options.theme ?? "dark";
const conds = String(options.conds).split(",").map((cond) => {
  const [label, port, copy = ""] = cond.split(":");
  return { label, port: Number(port), sheet: copy ? sheetOf(copy) : null };
});
const logFile = `${check}/results/${name}.log`;
const say = (line) => {
  console.log(line);
  appendFileSync(logFile, `${line}\n`);
};

const pause = 750;

async function shoot(page, clip) {
  await page.waitForTimeout(pause);
  return page.screenshot({ clip });
}

async function capture(browser, cond) {
  const { context, page, errors } = await open(browser, { port: cond.port, view, sheet: cond.sheet });
  await page.evaluate((value) => document.documentElement.setAttribute("data-theme", value), theme);
  await quiet(page, 500);
  const out = {};
  await page.keyboard.press("ArrowDown");
  await quiet(page, 250);
  await page.keyboard.press("ArrowDown");
  await quiet(page, 250);
  const nav = await page.getByRole("navigation", { name: "Screenshots" }).boundingBox();
  // The list and 24 px around it, so the bar on the edge of the main panel
  // and a focus ring outside a row are in the picture.
  const clip = {
    x: Math.max(0, nav.x - 24),
    y: Math.max(0, nav.y - 24),
    width: nav.width + 48,
    height: Math.min(900 - Math.max(0, nav.y - 24), nav.height + 48),
  };
  await page.mouse.move(900, 500);
  out.selected = await shoot(page, clip);
  // Keyboard focus on the selected row: a key first, so that the focus that
  // the script sets is a keyboard focus (:focus-visible).
  await page.keyboard.press("Shift");
  await page.evaluate(() => document.querySelector('a[data-row][aria-current="true"], a[data-row][tabindex="0"]')?.focus());
  await page.keyboard.press("ArrowDown");
  await quiet(page, 250);
  out["focus-row"] = await shoot(page, clip);
  const points = await page.evaluate(() => {
    const box = document.querySelector("a[data-row]").closest(".overflow-y-auto").getBoundingClientRect();
    const rows = [...document.querySelectorAll("a[data-row]")]
      .map((row) => ({ rect: row.getBoundingClientRect(), current: row.getAttribute("aria-current") === "true" }))
      .filter((row) => row.rect.top > box.top + 8 && row.rect.bottom < box.bottom - 8);
    const middle = (rect) => ({ x: rect.x + rect.width / 2, y: rect.y + rect.height / 2 });
    const plain = rows.filter((row) => !row.current);
    const selected = rows.find((row) => row.current);
    const first = plain[2] ?? plain[0];
    const next = plain[3] ?? plain[1];
    const button = [...document.querySelectorAll('nav[aria-label="Screenshots"] button[aria-expanded]')]
      .map((item) => item.getBoundingClientRect())
      .find((rect) => rect.top > box.top + 8 && rect.bottom < box.bottom - 8);
    return {
      row: first ? middle(first.rect) : null,
      gap: first && next ? { x: first.rect.x + first.rect.width / 2, y: (first.rect.bottom + next.rect.top) / 2, size: next.rect.top - first.rect.bottom } : null,
      selected: selected ? middle(selected.rect) : null,
      button: button ? middle(button) : null,
    };
  });
  if (points.row) {
    await page.mouse.move(points.row.x, points.row.y, { steps: 4 });
    out["hover-row"] = await shoot(page, clip);
  }
  if (points.gap && points.gap.size >= 1) {
    await page.mouse.move(points.gap.x, points.gap.y, { steps: 2 });
    out["hover-gap"] = await shoot(page, clip);
  }
  if (points.selected) {
    await page.mouse.move(points.selected.x, points.selected.y, { steps: 4 });
    out["hover-selected"] = await shoot(page, clip);
  }
  if (points.button) {
    await page.mouse.move(points.button.x, points.button.y, { steps: 4 });
    out["hover-button"] = await shoot(page, clip);
  }
  await page.mouse.move(900, 500, { steps: 4 });
  out["hover-out"] = await shoot(page, clip);
  if (view === "all") {
    await page.keyboard.press("Shift");
    await page.evaluate(() => {
      const box = document.querySelector("a[data-row]").closest(".overflow-y-auto").getBoundingClientRect();
      const button = [...document.querySelectorAll('nav[aria-label="Screenshots"] button[aria-expanded]')].find((item) => {
        const rect = item.getBoundingClientRect();
        return rect.top > box.top + 8 && rect.bottom < box.bottom - 8;
      });
      button?.focus({ preventScroll: true });
    });
    out["focus-button"] = await shoot(page, clip);
  }
  const state = await page.evaluate(() => ({
    gap: null,
    focus: document.activeElement?.tagName,
    focusVisible: document.activeElement?.matches(":focus-visible"),
  }));
  await context.close();
  return { out, clip, points, state, errors };
}

const browser = await chromium.launch({ channel: "chrome", headless: true });
say(`# ${name}: view=${view} theme=${theme} conds=${options.conds} chrome=${browser.version()} load ${load()}`);
const captures = {};
for (const cond of conds) {
  captures[cond.label] = await capture(browser, cond);
  for (const [state, buffer] of Object.entries(captures[cond.label].out)) {
    writeFileSync(`${check}/shots/${name}-${cond.label}-${state}.png`, buffer);
  }
  say(`${cond.label}: states ${Object.keys(captures[cond.label].out).join(", ")} | gap ${captures[cond.label].points.gap?.size} px | ${JSON.stringify(captures[cond.label].state)}${captures[cond.label].errors.length ? ` | errors ${JSON.stringify(captures[cond.label].errors).slice(0, 200)}` : ""}`);
}
const blank = await browser.newPage();
async function differ(first, second) {
  return blank.evaluate(
    async ([a, b]) => {
      const decode = async (data) => {
        const image = new Image();
        image.src = `data:image/png;base64,${data}`;
        await image.decode();
        const canvas = document.createElement("canvas");
        canvas.width = image.width;
        canvas.height = image.height;
        const context = canvas.getContext("2d");
        context.drawImage(image, 0, 0);
        return context.getImageData(0, 0, image.width, image.height);
      };
      const one = await decode(a);
      const two = await decode(b);
      if (one.width !== two.width || one.height !== two.height) return { differ: -1, total: one.width * one.height };
      let count = 0;
      let top = Infinity;
      let bottom = -1;
      let left = Infinity;
      let right = -1;
      for (let i = 0; i < one.data.length; i += 4) {
        const delta = Math.max(Math.abs(one.data[i] - two.data[i]), Math.abs(one.data[i + 1] - two.data[i + 1]), Math.abs(one.data[i + 2] - two.data[i + 2]));
        if (delta <= 2) continue;
        count += 1;
        const pixel = i / 4;
        const x = pixel % one.width;
        const y = Math.floor(pixel / one.width);
        top = Math.min(top, y);
        bottom = Math.max(bottom, y);
        left = Math.min(left, x);
        right = Math.max(right, x);
      }
      return { differ: count, total: one.width * one.height, box: count ? [left, top, right, bottom] : null };
    },
    [first.toString("base64"), second.toString("base64")],
  );
}
const reference = conds[0].label;
const table = {};
say(`\nPIXELS THAT DIFFER from "${reference}" (view=${view}, theme=${theme}); the box is left, top, right, bottom in the picture`);
for (const cond of conds.slice(1)) {
  table[cond.label] = {};
  for (const [state, buffer] of Object.entries(captures[cond.label].out)) {
    const base = captures[reference].out[state];
    if (!base) continue;
    const result = await differ(base, buffer);
    table[cond.label][state] = result;
    say(`${cond.label} | ${state} | ${result.differ} of ${result.total}${result.box ? ` | box ${result.box.join(",")}` : ""}`);
  }
}
await browser.close();
saveResult(`${name}.json`, { name, view, theme, conds: options.conds, table });
