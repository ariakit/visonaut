#!/bin/zsh
# Starts the production preview server of one copy.
# Usage: serve.sh <copy-name> <port>   (ports 4361 to 4364 only)
LANE="/Users/diegohaz/.claude/jobs/f65a6229/tmp/perf/record"
NAME="$1"
PORT="$2"
case "$PORT" in
  4361|4362|4363|4364) ;;
  *) echo "port $PORT is not a port of this role"; exit 1 ;;
esac
cd "$LANE/$NAME" || exit 1
exec node "$LANE/$NAME/node_modules/vite/bin/vite.js" preview --host 127.0.0.1 --port "$PORT" --strictPort
