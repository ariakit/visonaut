// Prints the scheduled invalidations of a result of invalidation.mjs, in
// the order of the trace, with the node, the change, and the set.
// Usage: node show-invalidation.mjs <result.json> [filter text]
import { readFileSync } from "node:fs";

const [file, filter] = process.argv.slice(2);
const data = JSON.parse(readFileSync(file, "utf8"));
const seen = new Set();
for (const item of data.scheduled) {
  const line = `${(item.nodeName ?? "?").slice(0, 70)} | class=${item.changedClass ?? ""} attr=${item.changedAttribute ?? ""} pseudo=${item.changedPseudo ?? ""} | set=${item.invalidationSet ?? ""} | ${item.invalidatedSelectorId ?? ""} | ${JSON.stringify(item.stackTrace ?? "").slice(0, 0)}`;
  if (filter && !line.includes(filter)) continue;
  if (seen.has(line)) continue;
  seen.add(line);
  console.log(line);
}
console.log("\nkeys of one scheduled event:", Object.keys(data.scheduled[0] ?? {}).join(", "));
console.log("\ninvalidator events:");
const seenInvalidator = new Set();
for (const item of data.invalidator) {
  const line = `${item.reason} | ${(item.nodeName ?? "").slice(0, 70)} | ${JSON.stringify(item.invalidationList ?? "")} | ${JSON.stringify(item.selectors ?? "").slice(0, 300)}`;
  if (filter && !line.includes(filter)) continue;
  if (seenInvalidator.has(line)) continue;
  seenInvalidator.add(line);
  console.log(line);
}
console.log("\nkeys of one invalidator event:", Object.keys(data.invalidator[0] ?? {}).join(", "));
