#!/bin/zsh
# Stops each server of this role (ports 4361 to 4364). Never port 4320.
# Usage: stop.sh [port ...]
PORTS=("$@")
if [[ ${#PORTS[@]} -eq 0 ]]; then
  PORTS=(4361 4362 4363 4364)
fi
for port in "${PORTS[@]}"; do
  case "$port" in
    4361|4362|4363|4364) ;;
    *) echo "port $port is not a port of this role"; continue ;;
  esac
  pids=$(lsof -nP -t -iTCP:$port -sTCP:LISTEN 2>/dev/null)
  if [[ -n "$pids" ]]; then
    echo "port $port: stop $pids"
    echo "$pids" | xargs kill
  else
    echo "port $port: no listener"
  fi
done
