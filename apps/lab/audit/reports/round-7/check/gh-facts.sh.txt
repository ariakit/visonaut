#!/bin/zsh
# GitHub facts of the new decisions. GET requests only.
echo "== visonaut: private, size, security settings"
gh api repos/ariakit/visonaut --jq '{private, size, visibility, admin: .permissions.admin, security: .security_and_analysis}'
echo "== visonaut private vulnerability reporting"
gh api repos/ariakit/visonaut/private-vulnerability-reporting --jq '.enabled'
echo "== ariakit: security settings"
gh api repos/ariakit/ariakit --jq '{private, security: .security_and_analysis}'
gh api repos/ariakit/ariakit/private-vulnerability-reporting --jq '.enabled'
echo "== SECURITY.md"
gh api repos/ariakit/visonaut/contents/SECURITY.md --silent; echo "visonaut SECURITY.md exit $?"
gh api repos/ariakit/visonaut/contents/.github/SECURITY.md --silent; echo "visonaut .github/SECURITY.md exit $?"
gh api repos/ariakit/ariakit/contents/SECURITY.md --silent; echo "ariakit SECURITY.md exit $?"
gh api repos/ariakit/ariakit/contents/.github/SECURITY.md --silent; echo "ariakit .github/SECURITY.md exit $?"
echo "== visonaut labels"
gh api repos/ariakit/visonaut/labels --paginate --jq '.[].name' | tr '\n' ','
echo
echo "== ariakit labels p0 p1 p2 easy hard"
gh api repos/ariakit/ariakit/labels --paginate --jq '.[] | select(.name=="p0" or .name=="p1" or .name=="p2" or .name=="easy" or .name=="hard") | .name' | tr '\n' ','
echo
echo "== visonaut issues (not pull requests): count, with labels, open"
gh api 'repos/ariakit/visonaut/issues?state=all&per_page=100' --paginate --jq '[.[] | select(.pull_request == null)] | {count: length, labeled: [.[] | select((.labels | length) > 0)] | length, open: [.[] | select(.state == "open")] | length}'
echo "== issue 204"
gh api repos/ariakit/visonaut/issues/204 --jq '{state, state_reason, closed_at}'
echo "== collaborators with push"
gh api 'repos/ariakit/visonaut/collaborators?per_page=100' --jq '{all: length, push: [.[] | select(.permissions.push)] | length}'
echo "== public members of the organization"
gh api 'orgs/ariakit/public_members?per_page=100' --jq 'length'
echo "== check runs of a7c8be2 in ariakit"
gh api 'repos/ariakit/ariakit/commits/a7c8be2/check-runs?per_page=100' --jq '{total: .total_count, submit: [.check_runs[] | select(.name | test("Visual Submit")) | .name]}'
echo "== pull request 237"
gh api repos/ariakit/visonaut/pulls/237 --jq '{title, merged_at, user: .user.login}'
echo "== pull request 104"
gh api repos/ariakit/visonaut/pulls/104 --jq '{title, merged_at}'
