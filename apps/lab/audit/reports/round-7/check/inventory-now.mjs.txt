// Counts the files that git would add below apps/lab today, and scans the files that the lane
// public-repository did not have in its inventory. Read only. Secret-like values are masked.
import { execFileSync } from "node:child_process";
import { readFileSync, writeFileSync } from "node:fs";
import { extname } from "node:path";
import { deflateSync } from "node:zlib";

const worktree = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const lane = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/public-repository";
const out = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/check-r9";

const files = execFileSync("git", ["ls-files", "--others", "--exclude-standard", "apps/lab"], {
  cwd: worktree,
  encoding: "utf8",
  maxBuffer: 64 * 1024 * 1024,
})
  .split("\n")
  .filter(Boolean);

const laneFiles = new Set(
  JSON.parse(readFileSync(`${lane}/inventory.json`, "utf8")).rows.map((row) => row.file),
);

function groupOf(file) {
  if (file.startsWith("apps/lab/audit/reports/")) return "audit/reports";
  if (file.startsWith("apps/lab/audit/")) return "audit (not reports)";
  if (file.startsWith("apps/lab/public/")) return "public";
  return "other";
}

const groups = {};
const textExtensions = new Set([
  ".json", ".md", ".txt", ".ts", ".html", ".tsx", ".jsonl", ".js", ".mjs", ".css", ".py", ".log",
  "", ".sh", ".jq", ".jsonc", ".svg",
]);
let textFiles = 0;
let pictures = 0;
let withLocalPath = 0;
const added = [];
for (const file of files) {
  const data = readFileSync(`${worktree}/${file}`);
  const key = groupOf(file);
  groups[key] ??= { files: 0, raw: 0, packed: 0 };
  groups[key].files += 1;
  groups[key].raw += data.length;
  groups[key].packed += deflateSync(data).length;
  const extension = extname(file);
  if (extension === ".png") pictures += 1;
  if (textExtensions.has(extension)) {
    textFiles += 1;
    if (data.includes("/Users/diegohaz")) withLocalPath += 1;
  }
  if (!laneFiles.has(file)) added.push(file);
}
const removed = [...laneFiles].filter((file) => !files.includes(file));

const mib = (bytes) => (bytes / 1048576).toFixed(1);
let total = { files: 0, raw: 0, packed: 0 };
for (const [key, value] of Object.entries(groups)) {
  total.files += value.files;
  total.raw += value.raw;
  total.packed += value.packed;
  console.log(`${key}: ${value.files} files, ${mib(value.raw)} MiB, deflated ${mib(value.packed)} MiB`);
}
console.log(`total: ${total.files} files, ${mib(total.raw)} MiB, deflated ${mib(total.packed)} MiB`);
console.log(`text files: ${textFiles}, PNG files: ${pictures}, text files with the local path: ${withLocalPath}`);
console.log(`files that the lane inventory did not have: ${added.length}`);
console.log(`files of the lane inventory that are gone: ${removed.length} ${removed.join(" ")}`);
const addedOutsideRoundSix = added.filter((file) => !file.startsWith("apps/lab/audit/reports/round-6/"));
console.log(`added outside reports/round-6: ${addedOutsideRoundSix.join(" ") || "none"}`);

const patterns = [
  ["github-token", /\b(?:gh[pousr]_[A-Za-z0-9]{20,}|github_pat_[A-Za-z0-9_]{20,})/g],
  ["private-key-block", /-----BEGIN [A-Z ]*PRIVATE KEY-----/g],
  ["jwt", /\beyJ[A-Za-z0-9_-]{8,}\.eyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}/g],
  ["aws-key", /\bAKIA[0-9A-Z]{16}\b/g],
  ["slack-token", /\bxox[baprs]-[A-Za-z0-9-]{10,}/g],
  ["npm-token", /\bnpm_[A-Za-z0-9]{30,}/g],
  ["key-with-sk", /\bsk-[A-Za-z0-9_-]{20,}/g],
  ["bearer-value", /\bBearer\s+[A-Za-z0-9._\-+/=]{20,}/g],
  [
    "keyword-assignment",
    /\b[A-Za-z0-9_]*(?:secret|passw(?:or)?d|api[_-]?key|private[_-]?key|access[_-]?token|refresh[_-]?token|client[_-]?secret)["']?\s*[:=]\s*["']?[A-Za-z0-9+/_\-.=]{16,}/gi,
  ],
  [
    "email",
    /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)*\.(?:com|org|net|io|dev|co|br|app|me)\b/g,
  ],
  ["cloudflare-account-id", /\baccount[_-]?id["']?\s*[:=]\s*["']?[a-f0-9]{32}\b/gi],
  [
    "cookie-value",
    /\b(?:__Secure-|__Host-)?[A-Za-z0-9_.-]*(?:session|auth|token)[A-Za-z0-9_.-]*=[A-Za-z0-9%._\-+/]{16,}/gi,
  ],
  [
    "token-keyword",
    /\b[A-Za-z0-9_]*(?:id[_-]?token|auth[_-]?token|session[_-]?token|webhook[_-]?secret)["']?\s*[:=]\s*["']?[A-Za-z0-9+/_\-.=]{16,}/gi,
  ],
];
const hits = [];
let scanned = 0;
for (const file of process.argv.includes("--all") ? files : added) {
  if (!textExtensions.has(extname(file))) continue;
  scanned += 1;
  const text = readFileSync(`${worktree}/${file}`, "utf8");
  for (const [kind, pattern] of patterns) {
    for (const match of text.matchAll(pattern)) {
      const value = match[0];
      const line = text.slice(0, match.index).split("\n").length;
      const shown = kind === "email" ? value : `${value.slice(0, 24)}...(length ${value.length})`;
      hits.push({ kind, file, line, shown });
    }
  }
}
console.log(`scanned ${scanned} added text files, hits: ${hits.length}`);
const byKind = {};
for (const hit of hits) {
  byKind[hit.kind] ??= new Map();
  byKind[hit.kind].set(hit.shown, (byKind[hit.kind].get(hit.shown) ?? 0) + 1);
}
for (const [kind, values] of Object.entries(byKind)) {
  console.log(`== ${kind}`);
  for (const [value, count] of values) console.log(`  ${count}x ${value}`);
}
writeFileSync(`${out}/inventory-now.json`, `${JSON.stringify({ groups, total, added, hits }, null, 1)}\n`);
