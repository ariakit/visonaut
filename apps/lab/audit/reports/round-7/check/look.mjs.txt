].join("");

const server = createServer((request, response) => {
  const send = (body) => {
    response.writeHead(200, { "content-type": "text/html; charset=utf-8" });
    response.end(body);
  };
  if (request.method !== "GET") {
    response.writeHead(405);
    response.end();
    return;
  }
  if (request.url?.startsWith("/artifact")) {
    send(`${hostSkeleton}${readFileSync(artifactFile, "utf8")}</body></html>`);
    return;
  }
  if (request.url?.startsWith("/standalone")) {
    send(readFileSync(standaloneFile, "utf8"));
    return;
  }
  response.writeHead(request.url === "/favicon.ico" ? 204 : 404);
  response.end();
});
await new Promise((resolve) => server.listen(4375, "127.0.0.1", resolve));

const results = [];
function expect(name, condition, detail = "") {
  results.push({ name, ok: Boolean(condition), detail });
  console.log(`${condition ? "pass" : "FAIL"}  ${name}${detail ? `  (${detail})` : ""}`);
}

const browser = await chromium.launch();

async function newPage({ width, height, touch }) {
  const context = await browser.newContext({
    viewport: { width, height },
    hasTouch: touch,
    isMobile: touch,
    colorScheme: "light",
    permissions: ["clipboard-read", "clipboard-write"],
  });
  const page = await context.newPage();
  const errors = [];
  page.on("pageerror", (error) => errors.push(`page error: ${error.message}`));
  page.on("console", (message) => {
    if (message.type() === "error") errors.push(`console: ${message.text()}`);
  });
  return { context, page, errors };
}

async function open(page, path = "/artifact") {
  await page.goto(`http://127.0.0.1:4375${path}`, { waitUntil: "load" });
  await page.locator("#doc[data-ready]").waitFor({ timeout: 15000 });
}

// 3. The look: each new panel and the plan part, at 1440 px and at 400 px.
const measures = [];
for (const viewport of [
  { name: "1440", width: 1440, height: 900, touch: false },
  { name: "400", width: 400, height: 800, touch: true },
]) {
  const { context, page, errors } = await newPage(viewport);
  await open(page);
  await page.waitForTimeout(800);
  for (const id of newIds) {
    const panel = page.locator(`[id="decision-${id}"]`);
    await panel.scrollIntoViewIfNeeded();
    await page.waitForTimeout(250);
    const box = await panel.boundingBox();
    const facts = await page.evaluate((panelId) => {
      const element = document.getElementById(panelId);
      const rect = element.getBoundingClientRect();
      let widest = 0;
      for (const child of element.querySelectorAll("*")) {
        widest = Math.max(widest, child.getBoundingClientRect().right);
      }
      return {
        pageOverflow: document.documentElement.scrollWidth - document.documentElement.clientWidth,
        right: rect.right,
        widestChild: widest,
        window: window.innerWidth,
      };
    }, `decision-${id}`);
    measures.push({ viewport: viewport.name, id, width: Math.round(box.width), height: Math.round(box.height), ...facts });
    await panel.screenshot({ path: join(shots, `${viewport.name}-${id}.png`) });
  }
  // The plan part: from its heading to the end of the section, in pieces of one window height.
  const plan = await page.evaluate(() => {
    const start = document.getElementById("your-answers-plan").getBoundingClientRect().top + window.scrollY;
    const section = document.getElementById("your-answers").getBoundingClientRect();
    return { start, end: section.bottom + window.scrollY };
  });
  let index = 0;
  const step = viewport.height - 120;
  for (let top = plan.start - 110; top < plan.end && index < 14; top += step) {
    await page.evaluate((value) => window.scrollTo(0, value), top);
    await page.waitForTimeout(250);
    index += 1;
    await page.screenshot({ path: join(shots, `${viewport.name}-plan-${String(index).padStart(2, "0")}.png`) });
  }
  // The open decisions table and the list of the 11 things.
  for (const anchor of ["your-answers-word", "your-answers-open", "your-answers-authorization"]) {
    await page.evaluate((id) => {
      const top = document.getElementById(id).getBoundingClientRect().top + window.scrollY;
      window.scrollTo(0, top - 110);
    }, anchor);
    await page.waitForTimeout(250);
    await page.screenshot({ path: join(shots, `${viewport.name}-${anchor}.png`) });
  }
  const overflow = await page.evaluate(() => {
    const wide = [];
    for (const table of document.querySelectorAll("#your-answers table, #access-trust table, #code-ci-docs table")) {
      const rect = table.getBoundingClientRect();
      const box = table.closest(".table-wrap, .scroll, [class*=scroll]") ?? table.parentElement;
      const parent = box.getBoundingClientRect();
      if (rect.width > window.innerWidth && parent.width > window.innerWidth) wide.push(table.caption?.textContent.trim().slice(0, 40));
    }
    return { page: document.documentElement.scrollWidth - document.documentElement.clientWidth, wide };
  });
  expect(`${viewport.name}: the page does not scroll sideways`, overflow.page <= 0, String(overflow.page));
  expect(`${viewport.name}: no table is wider than the window outside a scroll box`, overflow.wide.length === 0, overflow.wide.join(" | "));
  const outside = measures.filter((row) => row.viewport === viewport.name && row.widestChild > row.window + 1);
  expect(`${viewport.name}: no part of a new panel is right of the window`, outside.length === 0, outside.map((row) => row.id).join(" "));
  expect(`${viewport.name}: no page error`, errors.length === 0, errors.slice(0, 3).join(" | "));
  await context.close();
}
