// Searches the record for the held words of check-plan.mjs. It separates lines that are new in
// revision r9 from lines that revision r8 already had. Read only.
import { readFileSync, readdirSync } from "node:fs";
import { join } from "node:path";

const content =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content";
const backup = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/backup-r8";

const heldWords = [
  "get-access-token",
  "refresh-token",
  "get-session",
  "list-accounts",
  "update-user",
  "account-info",
  "requireSignature",
  "ipAddressHeaders",
  "cf-connecting-ip",
  "percent sequence",
  "change their own stored name",
  "Time Travel",
  "earlier databases",
  "earlier database",
  "X-Forwarded-For",
  "JWKS",
];

const files = [
  "decisions.json",
  "findings.json",
  "terms.json",
  ...readdirSync(join(content, "sections")).map((name) => `sections/${name}`),
];
const totals = {};
for (const file of files) {
  const now = readFileSync(join(content, file), "utf8").split("\n");
  const before = new Set(readFileSync(join(backup, file), "utf8").split("\n"));
  now.forEach((line, index) => {
    for (const word of heldWords) {
      if (!line.includes(word)) continue;
      const kind = before.has(line) ? "old" : "NEW";
      totals[`${kind} ${word}`] = (totals[`${kind} ${word}`] ?? 0) + 1;
      if (kind === "NEW" || process.argv.includes("--all")) {
        console.log(`${kind} ${file}:${index + 1} [${word}] ${line.trim().slice(0, 220)}`);
      }
    }
  });
}
console.log(JSON.stringify(totals, null, 1));
