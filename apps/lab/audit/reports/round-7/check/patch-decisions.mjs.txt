import { readFileSync, writeFileSync } from "node:fs";

const file =

const decisions = JSON.parse(readFileSync(file, "utf8"));

function target(id, field, optionId) {
  const decision = decisions.find((entry) => entry.id === id);
  if (!decision) throw new Error(`${id}: not found`);
  if (!optionId) return [decision, field];
  const option = decision.options.find((entry) => entry.id === optionId);
  if (!option) throw new Error(`${id}.${optionId}: not found`);
  return [option, field];
}

let count = 0;
function replace({ id, option, field, from, to }) {
  const [holder, key] = target(id, field, option);
  const parts = holder[key].split(from);
  if (parts.length !== 2) {
    throw new Error(`${id}${option ? `.${option}` : ""}.${field}: "${from.slice(0, 50)}" is there ${parts.length - 1} times`);
  }
  holder[key] = parts.join(to);
  count += 1;
}
