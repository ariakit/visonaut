// Counts of plan.json and findings-map.json for the plan part of section 15. Read only.
import { readFileSync } from "node:fs";

const folder = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/issues/final";
const plan = JSON.parse(readFileSync(`${folder}/plan.json`, "utf8"));
const map = JSON.parse(readFileSync(`${folder}/findings-map.json`, "utf8"));

console.log("== part waits");
const waits = {};
for (const wait of plan.partWaits) {
  waits[wait.issue] ??= [];
  waits[wait.issue].push(wait.waitsFor);
}
for (const [issue, list] of Object.entries(waits)) console.log(`  ${issue}: ${list.join(", ")}`);
const forContract = plan.partWaits.filter((wait) => wait.waitsFor === "contract");
console.log(
  `parts that wait for contract: ${forContract.length} in ${new Set(forContract.map((wait) => wait.issue)).size} issues`,
);
const dependOnContract = plan.issues.filter((issue) =>
  (issue.dependsOn ?? []).some((edge) => edge.key === "contract"),
);
console.log(`issues that depend on contract: ${dependOnContract.length} (${dependOnContract.map((issue) => issue.key).join(", ")})`);

console.log("== gates");
for (const gate of plan.gates) console.log(`  ${gate.issue}: ${gate.gate}`);

console.log("== lab pages");
for (const issue of plan.issues) {
  if ((issue.labPages ?? []).length > 0) console.log(`  ${issue.key}: ${issue.labPages.length} lab pages`);
}

console.log("== open decisions of the plan");
for (const entry of plan.openDecisions) {
  console.log(`  ${entry.id} | waits: ${entry.waits} | issues: ${entry.issues.join(", ")}`);
}
console.log("== filing decisions of the plan");
for (const entry of plan.filingDecisions) console.log(`  ${entry.id}: ${entry.question}`);

console.log("== contract pull requests");
const contract = plan.issues.find((issue) => issue.key === "contract");
console.log(contract.pullRequests.map((name, index) => `  ${index + 1}. ${name}`).join("\n"));
const rows = new Set();
for (const text of JSON.stringify(plan).matchAll(/MR-(\d+)/g)) rows.add(text[1]);
console.log(`map rows named in plan.json: ${[...rows].sort().join(" ")} (${rows.size})`);

console.log("== findings map");
console.log(`keys: ${Object.keys(map).join(", ")}`);
const entries = Array.isArray(map) ? map : (map.findings ?? Object.values(map));
const first = entries[0];
console.log(`first entry: ${JSON.stringify(first).slice(0, 500)}`);
const byDisposition = {};
const byReason = {};
for (const entry of entries) {
  byDisposition[entry.disposition] = (byDisposition[entry.disposition] ?? 0) + 1;
  const reason = `${entry.disposition} / ${entry.kindOfDisposition ?? entry.reasonKind ?? entry.group ?? entry.how ?? ""}`;
  byReason[reason] = (byReason[reason] ?? 0) + 1;
}
console.log(JSON.stringify(byDisposition));
console.log(JSON.stringify(byReason, null, 1));
