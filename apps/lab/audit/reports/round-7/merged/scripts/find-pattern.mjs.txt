// Prints each match of a regular expression with the text around it.
// Usage: node find-pattern.mjs <pattern> <file> [<file> ...]
import { readFileSync } from "node:fs";

const [pattern, ...files] = process.argv.slice(2);
const width = 110;

for (const file of files) {
  const text = readFileSync(file, "utf8");
  for (const match of text.matchAll(new RegExp(pattern, "g"))) {
    const line = text.slice(0, match.index).split("\n").length;
    const start = Math.max(0, match.index - width);
    const end = Math.min(text.length, match.index + match[0].length + width);
    console.log(`${file}:${line}: ...${text.slice(start, end).replaceAll("\n", " ")}...`);
  }
}
