// Prints decisions of the record by id. Usage: node show-decisions.mjs <path> <id> [<id> ...]
// With "--list", prints id, short, status of each decision.
import { readFileSync } from "node:fs";

const [file, ...ids] = process.argv.slice(2);
const data = JSON.parse(readFileSync(file, "utf8"));
const list = Array.isArray(data) ? data : data.decisions;
if (ids[0] === "--list") {
  for (const d of list) {
    console.log(`${d.id}\t${d.status}\t${d.short}`);
  }
  console.log(`count: ${list.length}`);
  console.log(`keys: ${Object.keys(list[0]).join(", ")}`);
} else {
  for (const id of ids) {
    const d = list.find((item) => item.id === id);
    if (!d) {
      console.log(`NOT FOUND: ${id}`);
      continue;
    }
    console.log(JSON.stringify(d, null, 2));
    console.log("-----");
  }
}
