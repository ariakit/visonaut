// Renumbers the list items below the heading "## Doubts" of notes.md, in order from 1.
import { readFileSync, writeFileSync } from "node:fs";

const file = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/limit-production/notes.md";
const lines = readFileSync(file, "utf8").split("\n");
const start = lines.findIndex((line) => line === "## Doubts");
if (start < 0) throw new Error("No heading '## Doubts'.");
let number = 0;
for (let index = start + 1; index < lines.length; index++) {
  const match = /^(\d+)\. (.*)$/.exec(lines[index]);
  if (!match) continue;
  number++;
  lines[index] = `${number}. ${match[2]}`;
}
writeFileSync(file, lines.join("\n"));
console.log(`doubts: ${number}`);
