import { readFileSync } from "node:fs";

const lane = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/limit-production";
const content =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content";
const files = [
  "decisions.json",
  "closed.md",
  "notes.md",
  "fragments/D-PRE-11.html",
  "fragments/closed-C1-data-inventory-outside.html",
  "fragments/closed-C2-access-app.html",
  "check/check-notes.md",
];
for (const file of files) {
  let text;
  try {
    text = readFileSync(`${lane}/${file}`, "utf8");
  } catch {
    console.log(`${file}: missing`);
    continue;
  }
  const emDash = [...text.matchAll(/—/g)].length;
  const enDash = [...text.matchAll(/–/g)].length;
  console.log(`${file}: em dashes ${emDash}, en dashes ${enDash}, chars ${text.length}`);
}

const decisions = JSON.parse(readFileSync(`${lane}/decisions.json`, "utf8"));
const reference = JSON.parse(readFileSync(`${content}/decisions.json`, "utf8"));
const findings = new Set(JSON.parse(readFileSync(`${content}/findings.json`, "utf8")).map((f) => f.id));
const keys = Object.keys(reference[0]).join(",");
const optionKeys = Object.keys(reference[0].options[0]).join(",");
for (const d of decisions) {
  console.log(`\n${d.id}: keys equal to the record: ${Object.keys(d).join(",") === keys}`);
  console.log(`  id is new: ${!reference.some((entry) => entry.id === d.id)}`);
  console.log(`  short ${d.short.length} chars, question ${d.question.length}, context ${d.context.length}`);
  console.log(`  recommended options: ${d.options.filter((o) => o.recommended).map((o) => o.id).join(",")}`);
  for (const o of d.options) {
    console.log(
      `  option ${o.id}: keys ok ${Object.keys(o).join(",") === optionKeys}, explanation ${o.explanation.length}, consequences ${o.consequences.length}`,
    );
  }
  console.log(`  unknown evidence: ${d.evidence.filter((id) => !findings.has(id)).join(",") || "none"}`);
  console.log(`  status ${d.status}, settledAnswer empty ${d.settledAnswer === ""}, rationale ${d.rationale.length}`);
}
