// Verify the drafts of this folder against the repository of today.
// Run: node scripts/verify.mjs
import { existsSync, readFileSync, readdirSync } from "node:fs";
import { join } from "node:path";

const worktree = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const folder = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/contract-future";
let failures = 0;

function fail(message) {
  failures += 1;
  console.log(`FAIL ${message}`);
}

function linesOf(file) {
  return readFileSync(join(worktree, file), "utf8").split("\n");
}

const cache = new Map();
function hasQuote(file, line, quote) {
  if (!cache.has(file)) cache.set(file, linesOf(file));
  const text = cache.get(file)[line - 1];
  return typeof text === "string" && text.includes(quote);
}

// 1. contract-changes.json: each quote is on its line.
const changes = JSON.parse(readFileSync(join(folder, "contract-changes.json"), "utf8"));
let quotes = 0;
for (const change of changes.changes) {
  quotes += 1;
  if (!hasQuote(change.file, change.line, change.today)) {
    fail(`${change.id}: the text is not on ${change.file}:${change.line}`);
  }
}
for (const row of changes.savedRequirements) {
  quotes += 1;
  if (!hasQuote(changes.verified.savedRequirements, row.line, row.today)) {
    fail(`saved requirement line ${row.line}: the text is not on that line: ${row.today}`);
  }
}
for (const row of changes.companionDocuments) {
  if (row.today === null) continue;
  quotes += 1;
  if (!hasQuote(row.file, row.line, row.today)) {
    fail(`${row.file}:${row.line}: the text is not on that line: ${row.today}`);
  }
}
console.log(`quotes checked: ${quotes}`);

// 2. The three classes hold each of the 58 decisions one time.
const decisions = JSON.parse(
  readFileSync(join(worktree, "apps/lab/audit/content/decisions.json"), "utf8"),
);
const classed = Object.values(changes.decisionClasses).flat();
const ids = decisions.map((decision) => decision.id);
for (const id of ids) {
  const count = classed.filter((entry) => entry === id).length;
  if (count !== 1) fail(`decision ${id} is in ${count} classes`);
}
for (const id of classed) {
  if (!ids.includes(id)) fail(`class entry ${id} is not a decision`);
}
console.log(`decisions in the record: ${ids.length}; classed: ${classed.length}`);
const notSettled = decisions.filter((decision) => decision.status !== "settled");
console.log(`decisions that are not settled: ${notSettled.length}`);

// 3. Each map row that a change or a saved requirement names exists.
const rowIds = new Set([...changes.mapRows.map((row) => row.id), changes.labRow.id, "NS-2"]);
for (const row of [...changes.savedRequirements, ...changes.earlierDecisions]) {
  for (const match of String(row.mapRow).matchAll(/(MR-\d+|NS-\d+)/g)) {
    if (!rowIds.has(match[1])) fail(`unknown map row ${match[1]}`);
  }
}

// 4. decisions.json of this folder: form and evidence.
const findings = JSON.parse(
  readFileSync(join(worktree, "apps/lab/audit/content/findings.json"), "utf8"),
);
const findingIds = new Set(findings.map((finding) => finding.id));
const drafts = JSON.parse(readFileSync(join(folder, "decisions.json"), "utf8"));
const keys = ["id", "short", "question", "context", "options", "evidence", "status", "settledAnswer", "rationale"];
for (const draft of drafts) {
  for (const key of keys) {
    if (!Object.hasOwn(draft, key)) fail(`${draft.id}: no field ${key}`);
  }
  if (ids.includes(draft.id)) fail(`${draft.id}: the id is in use`);
  if (draft.status !== "open") fail(`${draft.id}: status is not open`);
  if (draft.settledAnswer !== "") fail(`${draft.id}: settledAnswer is not empty`);
  if (!draft.rationale.startsWith("Needed: your choice.")) fail(`${draft.id}: rationale start`);
  const recommended = draft.options.filter((option) => option.recommended === true).length;
  if (recommended > 1) fail(`${draft.id}: ${recommended} recommended options`);
  const optionIds = new Set();
  for (const option of draft.options) {
    for (const key of ["id", "label", "explanation", "consequences", "recommended"]) {
      if (!Object.hasOwn(option, key)) fail(`${draft.id}/${option.id}: no field ${key}`);
    }
    if (optionIds.has(option.id)) fail(`${draft.id}: option ${option.id} two times`);
    optionIds.add(option.id);
  }
  for (const id of draft.evidence) {
    if (!findingIds.has(id)) fail(`${draft.id}: finding ${id} does not exist`);
  }
  const fragment = join(folder, "fragments", `${draft.id}.html`);
  if (!existsSync(fragment)) {
    fail(`${draft.id}: no fragment`);
  } else {
    const html = readFileSync(fragment, "utf8");
    const count = html.split(`<div data-decision="${draft.id}"></div>`).length - 1;
    if (count !== 1) fail(`${draft.id}: the fragment has ${count} placeholders`);
    for (const match of html.matchAll(/data-findings="([^"]+)"/g)) {
      for (const id of match[1].split(/\s+/)) {
        if (!findingIds.has(id)) fail(`${draft.id}: fragment finding ${id} does not exist`);
      }
    }
    for (const match of html.matchAll(/data-fact="([^"]+)"/g)) {
      if (!["measured", "counted", "estimate", "assumption"].includes(match[1])) {
        fail(`${draft.id}: unknown fact label ${match[1]}`);
      }
    }
    for (const match of html.matchAll(/href="([^"]+)"/g)) {
      if (!match[1].startsWith("#") && !match[1].startsWith("https://")) {
        fail(`${draft.id}: link ${match[1]} is not an anchor or https`);
      }
    }
    for (const match of html.matchAll(/class="language-([a-z]+)"/g)) {
      if (!["ts", "tsx", "sql", "json", "jsonc", "bash", "yaml", "text"].includes(match[1])) {
        fail(`${draft.id}: unknown code language ${match[1]}`);
      }
    }
  }
  console.log(`${draft.id}: ${draft.options.length} options, ${recommended} recommended, ${draft.evidence.length} findings`);
}

// 5. No em dash and no en dash in a draft of this folder.
function walk(directory) {
  const out = [];
  for (const entry of readdirSync(directory, { withFileTypes: true })) {
    const path = join(directory, entry.name);
    // The trial folder of the check holds copies of the record, which this lane did not write.
    if (entry.isDirectory() && entry.name === "trial") continue;
    if (entry.isDirectory()) out.push(...walk(path));
    else out.push(path);
  }
  return out;
}
for (const path of walk(folder)) {
  if (!/\.(md|json|html)$/.test(path)) continue;
  const text = readFileSync(path, "utf8");
  if (text.includes("—")) fail(`${path}: has an em dash`);
  if (/\.md$/.test(path)) {
    // A paragraph is one physical line: a prose line must not end in the middle of a sentence.
    const lines = text.split("\n");
    let inFence = false;
    lines.forEach((line, index) => {
      if (line.startsWith("```")) inFence = !inFence;
      if (inFence || line.trim() === "") return;
      const next = lines[index + 1] ?? "";
      const prose = (value) => /^[A-Za-z"'`(]/.test(value);
      if (prose(line) && prose(next)) fail(`${path}:${index + 1}: the paragraph has a line break`);
    });
  }
}

console.log(failures === 0 ? "OK: no failure" : `${failures} failures`);
process.exitCode = failures === 0 ? 0 : 1;
