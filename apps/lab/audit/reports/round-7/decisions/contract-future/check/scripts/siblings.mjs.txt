// Read-only helper. Prints the ids, the word counts, and the starts of the drafts of the other lanes.
import { existsSync, readFileSync } from "node:fs";

const base = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions";
const lanes = [
  "validator-logs",
  "limit-production",
  "documents-writes-rows",
  "public-repository",
  "contract-future",
];
const words = (text) => String(text || "").split(/\s+/).filter(Boolean).length;
for (const lane of lanes) {
  const file = `${base}/${lane}/decisions.json`;
  if (!existsSync(file)) continue;
  const drafts = JSON.parse(readFileSync(file, "utf8"));
  console.log(`\n## ${lane}`);
  for (const d of drafts) {
    console.log(
      `${d.id} | ${d.short} | ctx ${words(d.context)}w | ${d.options.length} opt: ${d.options
        .map((o) => `${o.id}${o.recommended ? "*" : ""} (${words(o.explanation)}+${words(o.consequences)}w)`)
        .join(", ")}`,
    );
    console.log(`  Q: ${d.question}`);
    console.log(`  rationale starts: ${String(d.rationale).slice(0, 90)}`);
  }
}
