// Read-only helper. Prints one line of the job timeline as text.
// Usage: node timeline-line.mjs <line number> [<from offset> <to offset>]
import { readFileSync } from "node:fs";

const file = "/Users/diegohaz/.claude/jobs/f65a6229/timeline.jsonl";
const lines = readFileSync(file, "utf8").split("\n").filter(Boolean);
const index = Number(process.argv[2]) - 1;
const from = Number(process.argv[3] || 0);
const to = Number(process.argv[4] || 100000);
const entry = JSON.parse(lines[index]);
console.log("keys:", Object.keys(entry).join(", "));
for (const [key, value] of Object.entries(entry)) {
  const text = typeof value === "string" ? value : JSON.stringify(value);
  console.log(`\n[${key}] length ${text.length}`);
  console.log(text.slice(from, to));
}
