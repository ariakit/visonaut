// Checks the drafts of this folder: JSON form, finding ids, one recommended option,
// no em dash, and each quoted file:line against the repository.
import { readFileSync, readdirSync } from "node:fs";

const folder = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/documents-writes-rows";
const root = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const backup = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round6/backup-r7";
let failures = 0;
const fail = (message) => {
  failures += 1;
  console.log("FAIL:", message);
};

const decisions = JSON.parse(readFileSync(`${folder}/decisions.json`, "utf8"));
const settled = JSON.parse(readFileSync(`${backup}/decisions.json`, "utf8"));
const liveSettled = JSON.parse(readFileSync(`${root}/apps/lab/audit/content/decisions.json`, "utf8"));
const findings = JSON.parse(readFileSync(`${backup}/findings.json`, "utf8"));
const liveFindings = JSON.parse(readFileSync(`${root}/apps/lab/audit/content/findings.json`, "utf8"));
const findingIds = new Set((Array.isArray(findings) ? findings : findings.findings).map((f) => f.id));
const liveFindingIds = new Set((Array.isArray(liveFindings) ? liveFindings : liveFindings.findings).map((f) => f.id));
const keys = Object.keys(settled[0]).join(",");
const optionKeys = Object.keys(settled[0].options[0]).join(",");
const usedIds = new Set([...settled, ...liveSettled].map((d) => d.id));

for (const d of decisions) {
  if (Object.keys(d).join(",") !== keys) fail(`${d.id}: keys differ: ${Object.keys(d)}`);
  if (usedIds.has(d.id)) fail(`${d.id}: id exists in the record`);
  if (!/^D-PRE-2[1-9]$/.test(d.id)) fail(`${d.id}: outside the id range`);
  if (d.status !== "open" || d.settledAnswer !== "") fail(`${d.id}: must be open with no answer`);
  if (!d.rationale.startsWith("Needed: your choice.")) fail(`${d.id}: rationale start`);
  if (!/It is wrong if|is wrong if/.test(d.rationale)) fail(`${d.id}: rationale does not say when it is wrong`);
  const recommended = d.options.filter((o) => o.recommended);
  if (recommended.length !== 1) fail(`${d.id}: ${recommended.length} recommended options`);
  const optionIds = new Set();
  for (const o of d.options) {
    if (Object.keys(o).join(",") !== optionKeys) fail(`${d.id}/${o.id}: option keys differ`);
    if (optionIds.has(o.id)) fail(`${d.id}/${o.id}: duplicate option id`);
    optionIds.add(o.id);
    console.log(`  ${d.id}/${o.id}${o.recommended ? " (recommended)" : ""}: explanation ${o.explanation.split(/\s+/).length} words, consequences ${o.consequences.split(/\s+/).length} words`);
  }
  for (const id of d.evidence) {
    if (!findingIds.has(id)) fail(`${d.id}: unknown finding ${id} in r7`);
    if (!liveFindingIds.has(id)) fail(`${d.id}: unknown finding ${id} in the live record`);
  }
  console.log(`${d.id}: context ${d.context.split(/\s+/).length} words, rationale ${d.rationale.split(/\s+/).length} words, ${d.options.length} options, evidence ${d.evidence.join(" ")}`);
}

const files = ["decisions.json", "closed.md", "notes.md", ...readdirSync(`${folder}/fragments`).map((f) => `fragments/${f}`)];
for (const file of files) {
  let text = "";
  try {
    text = readFileSync(`${folder}/${file}`, "utf8");
  } catch {
    console.log(`(not written yet: ${file})`);
    continue;
  }
  if (/—|–/.test(text)) fail(`${file}: has an em dash or an en dash`);
  if (file.endsWith(".html")) {
    const placeholders = [...text.matchAll(/<div data-decision="([^"]+)"><\/div>/g)].map((m) => m[1]);
    if (placeholders.length !== 1) fail(`${file}: ${placeholders.length} decision placeholders`);
    if (!decisions.some((d) => d.id === placeholders[0])) fail(`${file}: placeholder ${placeholders[0]} has no decision`);
    for (const match of text.matchAll(/data-findings="([^"]+)"/g)) {
      for (const id of match[1].split(/\s+/)) if (!liveFindingIds.has(id)) fail(`${file}: unknown finding ${id}`);
    }
    for (const match of text.matchAll(/class="language-([a-z]+)"/g)) {
      if (!["ts", "tsx", "sql", "json", "jsonc", "bash", "yaml", "text"].includes(match[1])) fail(`${file}: language ${match[1]}`);
    }
    for (const match of text.matchAll(/data-fact="([a-z]+)"/g)) {
      if (!["measured", "counted", "estimate", "assumption"].includes(match[1])) fail(`${file}: fact label ${match[1]}`);
    }
    // A paragraph must not hold a block element.
    for (const match of text.matchAll(/<p[\s>][\s\S]*?<\/p>/g)) {
      if (/<(div|ul|ol|table|pre|aside|h[1-6])[\s>]/.test(match[0])) fail(`${file}: a block inside a paragraph`);
    }
    const ids = [...text.matchAll(/ id="([^"]+)"/g)].map((m) => m[1]);
    const sections = readdirSync(`${root}/apps/lab/audit/content/sections`).map((f) => readFileSync(`${root}/apps/lab/audit/content/sections/${f}`, "utf8")).join("\n");
    for (const id of ids) if (sections.includes(`id="${id}"`)) fail(`${file}: anchor ${id} exists in the live record`);
    for (const match of text.matchAll(/href="#([^"]+)"/g)) {
      if (!sections.includes(`id="${match[1]}"`) && !match[1].startsWith("decision-")) fail(`${file}: link target ${match[1]} not found`);
    }
    console.log(`${file}: anchors ${ids.join(", ")}`);
  }
}
