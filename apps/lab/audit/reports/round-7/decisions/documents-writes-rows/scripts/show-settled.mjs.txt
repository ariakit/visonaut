// Prints the settled answer and the rationale of the named decisions (revision r7 copy).
import { readFileSync } from "node:fs";

const base = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round6/backup-r7";
const list = JSON.parse(readFileSync(`${base}/decisions.json`, "utf8"));
for (const id of process.argv.slice(2)) {
  const d = list.find((x) => x.id === id);
  console.log("=".repeat(80));
  console.log(d.id, "|", d.short);
  console.log("Q:", d.question);
  console.log("OPTIONS:", d.options.map((o) => `${o.id}${o.recommended ? "*" : ""}`).join(", "));
  console.log("EVIDENCE:", d.evidence.join(", "));
  console.log("ANSWER:", d.settledAnswer);
  console.log("RATIONALE:", d.rationale);
}
