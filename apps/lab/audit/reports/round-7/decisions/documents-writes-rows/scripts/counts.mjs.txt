// Counts that the drafts use. Each value is printed with its formula.
import { execFileSync } from "node:child_process";
import { readFileSync } from "node:fs";

const root = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const tracked = (dir) =>
  execFileSync("git", ["-C", root, "ls-files", dir], { encoding: "utf8" }).split("\n").filter(Boolean);
const words = (file) => readFileSync(`${root}/${file}`, "utf8").split(/\s+/).filter(Boolean).length;

const audit = tracked("docs/simplification-audit");
console.log("tracked files in docs/simplification-audit:", audit.length);
const operations = tracked("docs/operations");
console.log("tracked files in docs/operations:", operations.length);

const stay = [
  "docs/current-contract.md",
  "docs/review-guide.md",
  "docs/development.md",
  "docs/baseline-delta-cutover.md",
  "docs/operations/adapter-service-pins.md",
];
const stayWords = stay.map(words);
const total = stayWords.reduce((a, b) => a + b, 0);
const issue = words("docs/simplification-audit/contract-issue-1.md");
const cutover = words("docs/operations/simplification-cutover.md");
console.log("5 files that stay:", stayWords.join(" + "), "=", total);
console.log("contract-issue-1.md words:", issue, "| simplification-cutover.md words:", cutover);
console.log("stay + issue text, before step 9:", total + issue, "| after step 9:", total - stayWords[4] + issue);
console.log("stay after step 9 (4 files):", total - stayWords[4]);
console.log("ratio with issue text after step 9:", ((total - stayWords[4] + issue) / (total - stayWords[4])).toFixed(2));

console.log("\n# Webhook receipts");

  );
  console.log(`   share of 50 million at 4 rows: ${(((month * 4) / 50e6) * 100).toFixed(2)} % | USD with no included rows: ${((month * 4) / 1e6).toFixed(2)}`);
}
