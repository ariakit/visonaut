function checkDecisions(file, expectedIds) {
  const decisions = JSON.parse(readFileSync(`${folder}/${file}`, "utf8"));
  if (decisions.map((d) => d.id).join(",") !== expectedIds.join(",")) fail(`${file}: ids are ${decisions.map((d) => d.id)}`);
  for (const d of decisions) {
    if (Object.keys(d).join(",") !== keys) fail(`${d.id}: keys differ`);
    if (usedIds.has(d.id)) fail(`${d.id}: id exists in the record`);
    if (d.status !== "open" || d.settledAnswer !== "") fail(`${d.id}: must be open with no answer`);
    if (!d.rationale.startsWith("Needed: your choice.")) fail(`${d.id}: rationale start`);
    if (!/It is wrong if/.test(d.rationale)) fail(`${d.id}: rationale does not say when it is wrong`);
    if (d.options.filter((o) => o.recommended).length !== 1) fail(`${d.id}: not one recommended option`);
    if (d.options.length < 2) fail(`${d.id}: fewer than 2 options`);
    const ids = new Set();
    for (const o of d.options) {
      if (Object.keys(o).join(",") !== optionKeys) fail(`${d.id}/${o.id}: option keys differ`);
      if (ids.has(o.id)) fail(`${d.id}/${o.id}: duplicate option id`);
      ids.add(o.id);
      for (const field of ["id", "label", "explanation", "consequences"]) if (!o[field]) fail(`${d.id}/${o.id}: no ${field}`);
    }
    for (const id of d.evidence) if (!findingIds.has(id)) fail(`${d.id}: unknown finding ${id}`);
    const words = (text) => text.split(/\s+/).length;
    console.log(`${d.id}: context ${words(d.context)} words, rationale ${words(d.rationale)} words, options ${d.options.map((o) => `${o.id}${o.recommended ? "*" : ""} ${words(o.consequences)}w`).join(" | ")}`);
  }
  return decisions;
}

const panels = { "fragments/D-PRE-21.html": "D-PRE-21", "fragments/D-PRE-23.html": "D-PRE-23", "fragments/closed-index.html": null, "withdrawn/D-PRE-22.html": "D-PRE-22" };
for (const file of texts) {
  let text;
  try {
    text = readFileSync(`${folder}/${file}`, "utf8");
  } catch {
    console.log(`(not written yet: ${file})`);
    continue;
  }
  if (/[—–]/.test(text)) fail(`${file}: has an em dash or an en dash`);
  if (!file.endsWith(".html")) continue;
  const body = text.replace(/<!--[\s\S]*?-->/g, "");
  const found = [...body.matchAll(/<div data-decision="([^"]+)"><\/div>/g)].map((m) => m[1]);
  const expected = panels[file];
  if (expected === undefined) fail(`${file}: not a known fragment`);
  if (expected === null && found.length !== 0) fail(`${file}: must have no placeholder`);
  if (expected && (found.length !== 1 || found[0] !== expected)) fail(`${file}: placeholders ${found}`);
  if (/data-decision=/.test(body.replace(/<div data-decision="[^"]+"><\/div>/g, ""))) fail(`${file}: a placeholder that is not an empty div`);
  for (const match of body.matchAll(/data-findings="([^"]+)"/g)) {
    for (const id of match[1].split(/\s+/)) if (!findingIds.has(id)) fail(`${file}: unknown finding ${id}`);
  }
  for (const match of body.matchAll(/class="language-([a-z]+)"/g)) {
    if (!["ts", "tsx", "sql", "json", "jsonc", "bash", "yaml", "text"].includes(match[1])) fail(`${file}: language ${match[1]}`);
  }
  for (const match of body.matchAll(/data-fact="([a-z]+)"/g)) {
    if (!["measured", "counted", "estimate", "assumption"].includes(match[1])) fail(`${file}: fact label ${match[1]}`);
  }
  for (const match of body.matchAll(/<p[\s>][\s\S]*?<\/p>/g)) {
    if (/<(div|ul|ol|table|pre|aside|h[1-6])[\s>]/.test(match[0])) fail(`${file}: a block inside a paragraph`);
  }
  for (const match of body.matchAll(/href="([^"]+)"/g)) {
    const href = match[1];
    if (href.startsWith("#")) {
      if (!sections.includes(`id="${href.slice(1)}"`) && !href.startsWith("#decision-")) fail(`${file}: link target ${href} not found`);
    } else if (!href.startsWith("https://")) fail(`${file}: link ${href} is not https`);
  }
  if (/<(script|style|link|iframe|form)[\s>]/.test(body)) fail(`${file}: a forbidden element`);
  if (/style="[^"]*#[0-9a-fA-F]{3,6}/.test(body)) fail(`${file}: a color value`);
  const ids = [...body.matchAll(/ id="([^"]+)"/g)].map((m) => m[1]);
  for (const id of ids) {
    if (sections.includes(`id="${id}"`)) fail(`${file}: anchor ${id} exists in the record`);
    if (/^(decision-|finding-|findings-|term-definition-|code-copy-)/.test(id)) fail(`${file}: anchor ${id} has an engine prefix`);
  }
  console.log(`${file}: panel ${expected ?? "none"}, anchors ${ids.join(", ") || "none"}`);
}

}
