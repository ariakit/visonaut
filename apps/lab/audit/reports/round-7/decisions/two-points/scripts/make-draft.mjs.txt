// Makes a draft content folder from the stable copy of revision r7, with the proposed decision
// D-PRE-52 and its fragment in their place. Writes only below check/content of this part.
import { cpSync, mkdirSync, readFileSync, rmSync, writeFileSync } from "node:fs";

const base = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/two-points";
const source = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round6/backup-r7";
const target = `${base}/check/content`;

rmSync(target, { recursive: true, force: true });
mkdirSync(target, { recursive: true });
cpSync(source, target, { recursive: true });

const proposed = JSON.parse(readFileSync(`${base}/decisions.json`, "utf8"));
const decisions = JSON.parse(readFileSync(`${target}/decisions.json`, "utf8"));

// Document order: the new decision goes directly after the decision that its fragment follows.
const places = [
  ["D-PRE-52", "60-workspace-viewer.html", "D-WORK-03", "workspace-viewer-keys", "workspace-viewer-scope"],
];
for (const [id, , after] of places) {
  const index = decisions.findIndex((decision) => decision.id === after);
  if (index < 0) throw new Error(`${after} is not in the record`);
  decisions.splice(index + 1, 0, proposed.find((decision) => decision.id === id));
}
writeFileSync(`${target}/decisions.json`, `${JSON.stringify(decisions, null, 2)}\n`);

for (const [id, file, after, startAnchor, endAnchor] of places) {
  const anchor = `<div data-decision="${after}"></div>`;
  const fragment = readFileSync(`${base}/fragments/${id}.html`, "utf8").replace(
    /^<!--[\s\S]*?-->\n/,
    "",
  );
  const section = readFileSync(`${target}/sections/${file}`, "utf8");
  if (section.split(anchor).length !== 2) {
    throw new Error(`${file}: the anchor of ${id} is not in the file exactly one time`);
  }
  const start = section.indexOf(`id="${startAnchor}"`);
  const end = section.indexOf(`id="${endAnchor}"`);
  const at = section.indexOf(anchor);
  if (!(start >= 0 && start < at && at < end)) {
    throw new Error(`${file}: the place of ${id} is not between ${startAnchor} and ${endAnchor}`);
  }
  if (fragment.split(`data-decision="${id}"`).length !== 2) {
    throw new Error(`${id}: the fragment does not hold its placeholder exactly one time`);
  }
  // The fragment is indented as the section is: two spaces.
  const indented = fragment
    .trimEnd()
    .split("\n")
    .map((line) => (line ? `  ${line}` : line))
    .join("\n");
  writeFileSync(
    `${target}/sections/${file}`,
    section.replace(anchor, `${anchor}\n\n${indented}`),
  );
  console.log(`placed ${id} in ${file}, after ${after}, between #${startAnchor} and #${endAnchor}`);
}

// The text of the closed point has no panel. It goes directly before one heading.
{
  const file = "60-workspace-viewer.html";
  const before = '<h4 id="workspace-viewer-scope-place">';
  const fragment = readFileSync(`${base}/fragments/CLOSED-run-approval-large.html`, "utf8").replace(
    /^<!--[\s\S]*?-->\n/,
    "",
  );
  const section = readFileSync(`${target}/sections/${file}`, "utf8");
  if (section.split(before).length !== 2) {
    throw new Error(`${file}: the heading before which the closed text goes is not there one time`);
  }
  const indented = fragment
    .trimEnd()
    .split("\n")
    .map((line) => (line ? `  ${line}` : line))
    .join("\n");
  writeFileSync(
    `${target}/sections/${file}`,
    section.replace(before, `${indented.trimStart()}\n\n  ${before}`),
  );
  console.log(`placed the closed text in ${file}, before ${before}`);
  console.log(
    "closed text:",
    /[—–]/.test(fragment) ? "HAS A LONG DASH" : "no long dash",
    /data-decision=/.test(fragment) ? "HAS A PANEL" : "no panel",
  );
}

// Checks of the proposed decision that the build does not make.
const findings = JSON.parse(readFileSync(`${target}/findings.json`, "utf8"));
const known = new Set(findings.map((finding) => finding.id));
const record = JSON.parse(readFileSync(`${source}/decisions.json`, "utf8"));
const maximum = {
  short: Math.max(...record.map((decision) => decision.short.length)),
  question: Math.max(...record.map((decision) => decision.question.length)),
  context: Math.max(...record.map((decision) => decision.context.length)),
  consequences: Math.max(
    ...record.flatMap((decision) => decision.options.map((option) => option.consequences.length)),
  ),
  rationale: Math.max(...record.map((decision) => decision.rationale.length)),
};
console.log("largest text of the 58 decisions of the record:", JSON.stringify(maximum));
const keys = ["id", "short", "question", "context", "options", "evidence", "status", "settledAnswer", "rationale"];
const optionKeys = ["id", "label", "explanation", "consequences", "recommended"];
const banned = [/[—–]/, /\bvery\b/i, /\butilize/i, /\bleverage/i, /\bvia\b/i];
for (const decision of proposed) {
  const text = JSON.stringify(decision);
  const problems = [];
  if (JSON.stringify(Object.keys(decision)) !== JSON.stringify(keys)) problems.push("keys");
  if (decision.status !== "open" || decision.settledAnswer !== "") problems.push("status");
  if (!decision.rationale.startsWith("Needed: your choice.")) problems.push("rationale start");
  if (decision.options.filter((option) => option.recommended).length > 1) {
    problems.push("recommended count");
  }
  if (new Set(decision.options.map((option) => option.id)).size !== decision.options.length) {
    problems.push("option ids repeat");
  }
  if (record.some((entry) => entry.id === decision.id)) problems.push("id is in the record");
  for (const id of decision.evidence) if (!known.has(id)) problems.push(`finding ${id}`);
  for (const pattern of banned) if (pattern.test(text)) problems.push(`text ${pattern}`);
  for (const option of decision.options) {
    if (JSON.stringify(Object.keys(option)) !== JSON.stringify(optionKeys)) {
      problems.push(`option keys ${option.id}`);
    }
  }
  const sizes = {
    short: decision.short.length,
    question: decision.question.length,
    context: decision.context.length,
    consequences: Math.max(...decision.options.map((option) => option.consequences.length)),
    rationale: decision.rationale.length,
  };
  for (const name of Object.keys(maximum)) {
    if (sizes[name] > maximum[name]) problems.push(`${name} is longer than each one of the record`);
  }
  console.log(decision.id, JSON.stringify(sizes), problems.length ? problems.join(", ") : "form is correct");
}
for (const [id] of places) {
  const fragment = readFileSync(`${base}/fragments/${id}.html`, "utf8");
  const findingIds = [...fragment.matchAll(/data-findings="([^"]+)"/g)].flatMap((match) =>
    match[1].split(/\s+/),
  );
  const unknown = findingIds.filter((finding) => !known.has(finding));
  console.log(
    id,
    "fragment:",
    /[—–]/.test(fragment) ? "HAS A LONG DASH" : "no long dash",
    unknown.length ? `unknown findings ${unknown.join(" ")}` : `${findingIds.length} finding ids exist`,
  );
}
for (const name of ["closed.md", "notes.md"]) {
  try {
    const text = readFileSync(`${base}/${name}`, "utf8");
    console.log(name, /[—–]/.test(text) ? "HAS A LONG DASH" : "no long dash");
  } catch {
    console.log(name, "is not written");
  }
}
