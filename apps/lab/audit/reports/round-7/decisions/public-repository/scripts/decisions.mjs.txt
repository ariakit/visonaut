// Prints the settled decisions. Usage: node decisions.mjs            (id, short, settled answer, evidence)
//                                        node decisions.mjs ID [ID...] (the complete decision)
import { readFileSync } from "node:fs";

const base =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content";
const decisions = JSON.parse(readFileSync(`${base}/decisions.json`, "utf8"));
const ids = process.argv.slice(2);
if (ids.length === 0) {
  for (const d of decisions) {
    console.log(`${d.id} | ${d.short} | ${d.question}\n   -> ${d.settledAnswer.slice(0, 420)}\n   evidence: ${d.evidence.join(" ")}`);
  }
} else {
  for (const id of ids) {
    console.log(JSON.stringify(decisions.find((d) => d.id === id), null, 1));
  }
}
