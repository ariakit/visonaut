// Reads scan-hits.json and prints the secret-like hits (masked) with a shape test, and the complete host list.
import { readFileSync } from "node:fs";

const out = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/public-repository";
const worktree = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const data = JSON.parse(readFileSync(`${out}/scan-hits.json`, "utf8"));

function entropy(value) {
  const count = {};
  for (const ch of value) count[ch] = (count[ch] ?? 0) + 1;
  let bits = 0;
  for (const n of Object.values(count)) {
    const p = n / value.length;
    bits -= p * Math.log2(p);
  }
  return bits;
}

console.log("== secret-like hits (value masked; shape test on the real value)");
for (const hit of data.hits) {
  const text = readFileSync(`${worktree}/${hit.file}`, "utf8").split("\n")[hit.line - 1] ?? "";
  const m = /[:=]\s*["']?([A-Za-z0-9+/_\-.=%]{16,})/.exec(text.slice(Math.max(0, text.indexOf(hit.shown.slice(0, 6)))));
  const value = m ? m[1] : "";
  const words = /^[a-z0-9]+([-_.][a-z0-9]+)+$/i.test(value);
  console.log(
    `  ${hit.kind} | ${hit.file}:${hit.line} | ${hit.shown} | hyphenated-words=${words} | entropy=${value ? entropy(value).toFixed(2) : "n/a"}`,
  );
}

console.log("\n== all hosts");
for (const v of data.summary["url-host"]) {
  console.log(`  ${String(v.count).padStart(5)}x ${String(v.files).padStart(4)} files  ${v.value}  [${v.sample[0]}]`);
}
console.log("\n== all uuids");
for (const v of data.summary.uuid) {
  console.log(`  ${String(v.count).padStart(5)}x ${String(v.files).padStart(4)} files  ${v.value}  [${v.sample[0]}]`);
}
