// Lists what a branch with apps/lab would publish: the files that git would add (untracked and not ignored).
// Read-only for the worktree. Writes inventory.json next to this script folder.
import { execFileSync } from "node:child_process";
import { statSync, writeFileSync } from "node:fs";
import { extname } from "node:path";

const worktree = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const out = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/decisions/public-repository";

const raw = execFileSync(
  "git",
  ["-C", worktree, "ls-files", "--others", "--exclude-standard", "-z", "--", "apps/lab"],
  { maxBuffer: 1 << 28 },
).toString("utf8");
const files = raw.split("\0").filter(Boolean);

const tracked = execFileSync("git", ["-C", worktree, "ls-files", "-z", "--", "apps/lab"], {
  maxBuffer: 1 << 28,
})
  .toString("utf8")
  .split("\0")
  .filter(Boolean);

const rows = [];
for (const file of files) {
  let size = 0;
  try {
    size = statSync(`${worktree}/${file}`).size;
  } catch {
    continue;
  }
  rows.push({ file, size, ext: extname(file).toLowerCase() || "(none)" });
}

const byExt = {};
for (const row of rows) {
  byExt[row.ext] ??= { count: 0, bytes: 0 };
  byExt[row.ext].count += 1;
  byExt[row.ext].bytes += row.size;
}
const byTop = {};
for (const row of rows) {
  const parts = row.file.split("/");
  const key = parts.slice(0, 4).join("/");
  byTop[key] ??= { count: 0, bytes: 0 };
  byTop[key].count += 1;
  byTop[key].bytes += row.size;
}

const total = rows.reduce((sum, row) => sum + row.size, 0);
console.log("tracked files below apps/lab:", tracked.length);
console.log("files that git would add:", rows.length, "bytes:", total, `(${(total / 1048576).toFixed(1)} MiB)`);
console.log("\nby extension:");
for (const [ext, v] of Object.entries(byExt).sort((a, b) => b[1].bytes - a[1].bytes)) {
  console.log(`  ${ext.padEnd(10)} ${String(v.count).padStart(5)} files ${String(v.bytes).padStart(11)} bytes`);
}
console.log("\nby folder (4 levels):");
for (const [key, v] of Object.entries(byTop).sort((a, b) => b[1].bytes - a[1].bytes)) {
  console.log(`  ${String(v.count).padStart(5)} files ${String(v.bytes).padStart(11)} bytes  ${key}`);
}
console.log("\n25 largest files:");
for (const row of [...rows].sort((a, b) => b.size - a.size).slice(0, 25)) {
  console.log(`  ${String(row.size).padStart(10)}  ${row.file}`);
}
writeFileSync(`${out}/inventory.json`, `${JSON.stringify({ total, count: rows.length, byExt, byTop, rows }, null, 1)}\n`);
