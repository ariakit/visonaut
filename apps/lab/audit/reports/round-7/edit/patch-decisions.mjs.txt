// Corrections of revision r9 in content/decisions.json. Each change names its
// decision and its field, and each old text must stand exactly one time in that
// field. An earlier decision changes only in its rationale, except the one
// login of D-RES-05, which the task names.
//
// Usage: node patch-decisions.mjs
import { readFileSync, writeFileSync } from "node:fs";

const file =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content/decisions.json";
const decisions = JSON.parse(readFileSync(file, "utf8"));

function find(id) {
  const decision = decisions.find((item) => item.id === id);
  if (!decision) throw new Error(`${id} is not in decisions.json.`);
  return decision;
}

function replaceOnce(text, from, to, where) {
  const count = text.split(from).length - 1;
  if (count !== 1) throw new Error(`${where}: the old text stands ${count} time(s).`);
  return text.replace(from, () => to);
}

function rationale(id, from, to) {
  const decision = find(id);
  decision.rationale = replaceOnce(decision.rationale, from, to, `${id} rationale`);
}

function append(id, sentence) {
  const decision = find(id);
  if (decision.rationale.includes(sentence)) {
    throw new Error(`${id} rationale already has the sentence.`);
  }
  decision.rationale = `${decision.rationale} ${sentence}`;
}

function option(id, optionId, field, from, to) {
  const decision = find(id);
  const item = decision.options.find((entry) => entry.id === optionId);
  if (!item) throw new Error(`${id} has no option ${optionId}.`);
  item[field] = replaceOnce(item[field], from, to, `${id} option ${optionId} ${field}`);
}
