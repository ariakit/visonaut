// Inserts one fragment file, without its first comment and with the indentation of a
// section, before the line that holds a marker. The marker must stand exactly one time.
//
// Usage: node insert-fragment.mjs <section file name> <marker> <fragment file>
import { readFileSync, writeFileSync } from "node:fs";

const sections =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/content/sections";

const [name, marker, fragmentFile] = process.argv.slice(2);
const file = `${sections}/${name}`;
const lines = readFileSync(file, "utf8").split("\n");

const indexes = [];
for (const [index, line] of lines.entries()) {
  if (line.includes(marker)) indexes.push(index);
}
if (indexes.length !== 1) {
  throw new Error(`${name}: the marker ${marker} stands ${indexes.length} time(s).`);
}

const text = readFileSync(fragmentFile, "utf8").replace(/^\s*<!--[\s\S]*?-->\s*\n/, "");
const fragment = text.replace(/\s+$/, "").split("\n");
const indents = fragment
  .filter((line) => line.trim() !== "")
  .map((line) => line.match(/^ */)[0].length);
const shift = 2 - Math.min(...indents);
const shifted = fragment.map((line) => {
  if (line.trim() === "") return "";
  return shift >= 0 ? " ".repeat(shift) + line : line.slice(-shift);
});

lines.splice(indexes[0], 0, ...shifted, "");
writeFileSync(file, lines.join("\n"));
console.log(`${name}: ${shifted.length} lines in, before ${marker}`);
