for (const [name, note] of Object.entries(notes)) {
  const file = `${sections}/${name}`;
  const text = readFileSync(file, "utf8");
  if (text.includes(note.slice(0, 60))) {
    console.log(`${name}: the note is already there.`);
    continue;
  }
  const sectionStart = text.indexOf("<section");
  const commentEnd = text.lastIndexOf("-->", sectionStart);
  let next;
  if (commentEnd === -1) {
    next = `<!--\n  ${note}\n-->\n${text}`;
  } else {
    next = `${text.slice(0, commentEnd).replace(/\s+$/, "")}\n  ${note}\n${text.slice(commentEnd)}`;
  }
  writeFileSync(file, next);
  console.log(`${name}: note added.`);
}
