/** One Miniflare instance with several empty D1 databases. Each scenario group takes one. */
export function createRuntime(databaseCount: number) {
  const names = Array.from({ length: databaseCount }, (_, index) => `DB${index}`);
  const runtime = new Miniflare(
    convertV4MiniflareOptions({
      modules: true,
      script: "export default { fetch() { return new Response('ok'); } }",
      compatibilityDate: "2026-09-22",
      d1Databases: names,
      r2Buckets: ["IMAGES", "QUARANTINE"],
    }),
  );
  let next = 0;
  return {
    runtime,
    async database() {
      const name = names[next];
      next += 1;
      if (!name) throw new Error("The probe needs more D1 databases.");
      return runtime.getD1Database(name);
    },
  };
}

interface BatchRecord {
  payloadBytes: number;
  statements: number;
  writes: number;
  localMs: number;
  marker: string;
}

interface MeterEntry {
  sql: string;
  rowsWritten: number;
  rowsRead: number;
}

interface RaceRule {
  pattern: RegExp;
  action: () => Promise<void>;
}

interface FaultRule {
  pattern: RegExp;
  remaining: number;
  message: string;
}

/**
 * Wraps D1 so that a probe can run one action immediately before a matching write
 * batch (a deterministic race), or make a matching batch throw (a storage fault).
 */
export function hookDatabase(native: any) {
  const sqlOf = new WeakMap<object, string>();
  const bytesOf = new WeakMap<object, number>();
  const nativeOf = new WeakMap<object, any>();
  const state = {
    race: null as RaceRule | null,
    fault: null as FaultRule | null,
    batches: [] as BatchRecord[],
    singles: 0,
    meter: { calls: 0, statements: 0, rowsWritten: 0, rowsRead: 0, log: [] as MeterEntry[] },
  };
  const record = (sql: string, meta: any) => {
    state.meter.statements += 1;
    state.meter.rowsWritten += Number(meta?.rows_written ?? 0);
    state.meter.rowsRead += Number(meta?.rows_read ?? 0);
    state.meter.log.push({
      sql: sql.replace(/\s+/gu, " ").slice(0, 90),
      rowsWritten: Number(meta?.rows_written ?? 0),
      rowsRead: Number(meta?.rows_read ?? 0),
    });
  };
  const wrap = (statement: any, sql: string, values: unknown[] = []): any => {
    const proxy = new Proxy(statement, {
      get(target, key) {
        if (key === "bind") {
          return (...bound: unknown[]) => wrap(target.bind(...bound), sql, bound);
        }
        const value = Reflect.get(target, key);
        if (typeof value !== "function") return value;
        if (key === "first") {
          // D1 gives no meta for first(). all() gives the same row and the meta.
          return async (column?: string) => {
            state.singles += 1;
            state.meter.calls += 1;
            const result = await target.all();
            record(sql, result.meta);
            const row = result.results?.[0] ?? null;
            if (column == null) return row;
            return row ? (row[column] ?? null) : null;
          };
        }
        if (key === "all" || key === "run") {
          return async (...values: unknown[]) => {
            state.singles += 1;
            state.meter.calls += 1;
            const result = await value.apply(target, values);
            record(sql, result?.meta);
            return result;
          };
        }
        if (key === "raw") {
          return (...values: unknown[]) => {
            state.singles += 1;
            state.meter.calls += 1;
            state.meter.statements += 1;
            return value.apply(target, values);
          };
        }
        return value.bind(target);
      },
    });
    sqlOf.set(proxy, sql);
    bytesOf.set(proxy, sql.length + values.reduce((sum: number, value) => sum + String(value ?? "").length, 0));
    nativeOf.set(proxy, statement);
    return proxy;
  };
  const database = new Proxy(native, {
    get(target, key) {
      if (key === "prepare") return (sql: string) => wrap(target.prepare(sql), sql);
      if (key === "batch") {
        return async (list: object[]) => {
          const sqls = list.map((entry) => sqlOf.get(entry) ?? "");
          const race = state.race;
          if (race && sqls.some((sql) => race.pattern.test(sql))) {
            state.race = null;
            await race.action();
          }
          const fault = state.fault;
          if (fault && fault.remaining > 0 && sqls.some((sql) => fault.pattern.test(sql))) {
            fault.remaining -= 1;
            throw new Error(fault.message);
          }
          const started = performance.now();
          try {
            state.meter.calls += 1;
            const results = await target.batch(list.map((entry) => nativeOf.get(entry)));
            results.forEach((result: any, index: number) => record(sqls[index] ?? "", result?.meta));
            return results;
          } finally {
            const writes = sqls.filter((sql) => !/^\s*SELECT/iu.test(sql));
            state.batches.push({
              payloadBytes: list.reduce((sum, entry) => sum + (bytesOf.get(entry) ?? 0), 0),
              statements: list.length,
              writes: writes.length,
              localMs: Math.round((performance.now() - started) * 10) / 10,
              marker:
                sqls.find((sql) => /INSERT INTO visonaut_commands|work_tasks/iu.test(sql))?.slice(
                  0,
                  60,
                ) ?? (writes[0] ?? sqls[0] ?? "").replace(/\s+/gu, " ").slice(0, 60),
            });
          }
        };
      }
      const value = Reflect.get(target, key);
      return typeof value === "function" ? value.bind(target) : value;
    },
  });
  return { database, state };
}

export function sha(name: string) {
  return createHash("sha1").update(name).digest("hex");
}

        }
        if (method === "PATCH") {
          github.log.push(`PATCH check ${checkId} -> ${body.conclusion ?? body.status}`);
          Object.assign(checks.get(checkId) ?? {}, body);
          return {};
        }
        if (path.includes("/commits/")) {
          const commit = /commits\/([a-f0-9]{40})/u.exec(path)?.[1];
          github.log.push("GET commit checks");
          return { check_runs: [...checks.values()].filter((check) => check.head_sha === commit) };
        }
        github.log.push(`GET check ${checkId}`);
        if (github.failGet) throw new Error("GitHub read failed.");
        return checks.get(checkId);
      },
    },
  };

  return {
    native,
    hooked,
    service,
    bindings,
    sent,
    users,
    api,
    flush,
    reviewSession,
    fixture,
    promote,
    id,
    rows,
    row,
    verdicts,
    status,
    counts,
    changed,
    command,
    commandPath,
    operations,
    github,
    checks,
    githubLog,
  };
}

export type World = Awaited<ReturnType<typeof createWorld>>;

/** Keeps only the parts of an HTTP result that the report needs. */
export function shape(result: { status: number; body: any; retryAfter?: string | null }) {
  const body = result.body && typeof result.body === "object" ? result.body : {};
  return {
    status: result.status,
    keys: Object.keys(body),
    ...(body.error ? { error: body.error } : {}),
    ...(body.reviewer ? { reviewer: body.reviewer } : {}),
    ...(body.queued ? { queued: body.queued } : {}),
    ...(body.runStatus ? { runStatus: body.runStatus } : {}),
    ...(body.noop ? { noop: body.noop } : {}),
    ...(body.revisions ? { revisions: body.revisions.length } : {}),
    ...(body.model ? { modelRunStatus: body.model.run?.status } : {}),
    ...(result.retryAfter ? { retryAfter: result.retryAfter } : {}),
  };
}
