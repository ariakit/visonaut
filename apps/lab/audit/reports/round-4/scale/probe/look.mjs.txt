// Opens the built draft in Chrome at 1440 px and at 400 px, takes pictures of
// the new part, uses its calculator, and records each state as text.
import { mkdirSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";

const root = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const lane = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round4/scale";
const shots = `${lane}/shots`;
const require = createRequire(`${root}/package.json`);
const { chromium } = require("@playwright/test");
const url = `file://${lane}/out/index.html`;
mkdirSync(shots, { recursive: true });

const report = [];
const browser = await chromium.launch({ channel: "chrome" });
for (const [width, height] of [
  [1440, 900],
  [400, 800],
]) {
  const context = await browser.newContext({ viewport: { width, height }, hasTouch: width < 600 });
  const page = await context.newPage();
  const errors = [];
  page.on("console", (message) => {
    if (message.type() === "error") errors.push(message.text());
  });
  page.on("pageerror", (error) => errors.push(String(error)));
  await page.goto(url, { waitUntil: "load" });
  await page.waitForSelector("#prompt-text");
  await page.waitForTimeout(500);
  const jump = async (selector, offset = 80) => {
    await page.evaluate(
      ([target, top]) => {
        const element = document.querySelector(target);
        if (!element) throw new Error(`missing ${target}`);
        window.scrollTo(0, element.getBoundingClientRect().top + window.scrollY - top);
      },
      [selector, offset],
    );
    await page.waitForTimeout(250);
  };
  const shot = async (name) => page.screenshot({ path: `${shots}/${width}-${name}.png` });
  const demo = page.locator("#demo-data-scale-run");
  const demoText = async () => (await demo.innerText()).replace(/\n+/gu, " | ");
  const setSlider = async (index, value) => {
    await demo
      .locator('input[type="range"]')
      .nth(index)
      .evaluate((input, next) => {
        input.value = String(next);
        input.dispatchEvent(new Event("input", { bubbles: true }));
        input.dispatchEvent(new Event("change", { bubbles: true }));
      }, value);
    await page.waitForTimeout(150);
  };

  await jump("#data-scale-run");
  await shot("part-1");
  // The part is tall: take a picture of each screen down to the next heading.
  const partHeight = await page.evaluate(() => {
    const start = document.querySelector("#data-scale-run");
    const end = document.querySelector("#data-read-path");
    return end.getBoundingClientRect().top - start.getBoundingClientRect().top;
  });
  const screens = Math.ceil(partHeight / (height - 140));
  for (let screen = 2; screen <= screens; screen++) {
    await page.mouse.wheel(0, height - 140);
    await page.waitForTimeout(200);
    await shot(`part-${screen}`);
  }

  await jump("#demo-data-scale-run");
  const start = await demoText();
  await shot("calculator-start");
  await setSlider(0, 100000);
  await setSlider(1, 0.1);
  await setSlider(2, 15);
  const small = await demoText();
  await setSlider(0, 500000);
  await setSlider(1, 10);
  await setSlider(2, 23);
  const large = await demoText();
  await shot("calculator-large");
  await demo.getByRole("button", { name: /reset/iu }).click();
  await page.waitForTimeout(150);
  const afterReset = await demoText();

  const overflow = await page.evaluate(
    () => document.documentElement.scrollWidth - document.documentElement.clientWidth,
  );
  const table = await page.evaluate(() => {
    const heading = document.querySelector("#data-scale-run");
    let node = heading.nextElementSibling;
    while (node && !node.querySelector?.("table") && node.tagName !== "TABLE") {
      node = node.nextElementSibling;
    }
    const element = node?.tagName === "TABLE" ? node : node?.querySelector("table");
    const box = element?.parentElement;
    return {
      rows: element?.querySelectorAll("tbody tr").length,
      tableWidth: element?.scrollWidth,
      boxWidth: box?.clientWidth,
      factLabels: element?.querySelectorAll("[data-fact]").length,
    };
  });
  const anchors = await page.evaluate(() =>
    ["#data-scale-run", "#data-scale-verdicts", "#data-scale-order", "#demo-data-scale-run"].map(
      (selector) => [selector, document.querySelector(selector) !== null],
    ),
  );
  report.push({
    width,
    errors,
    overflow,
    screens,
    table,
    anchors,
    resetGivesStart: afterReset === start,
    calculator: { start, small, large },
  });
  await context.close();
}
await browser.close();
writeFileSync(`${lane}/results/look.json`, `${JSON.stringify(report, null, 2)}\n`);
console.log(JSON.stringify(report, null, 2));
