const results = [];
function expect(condition, message) {
  if (!condition) {
    throw new Error(message);
  }
}
async function check(name, run) {
  try {
    const note = await run();
    results.push({ name, ok: true, note });
    console.log(`pass  ${name}${note ? `  (${note})` : ""}`);
  } catch (error) {
    results.push({ name, ok: false, note: error.message });
    console.log(`FAIL  ${name}  (${error.message.split("\n")[0]})`);
  }
}

async function newPage({ width, height, touch = false }) {
  const context = await browser.newContext({
    viewport: { width, height },
    hasTouch: touch,
    isMobile: touch,
    colorScheme: "dark",
    permissions: ["clipboard-read", "clipboard-write"],
  });
  const page = await context.newPage();
  const problems = [];
  page.on("console", (message) => {
    if (message.type() === "error" && !message.text().includes("fonts.g")) {
      problems.push(`console: ${message.text()}`);
    }
  });
  page.on("pageerror", (error) => problems.push(`page error: ${error.message}`));
  const open = async () => {
    await page.goto(url, { waitUntil: "load" });
    await page.locator("#doc[data-ready]").waitFor({ timeout: 15000 });
  };
  const stored = () => {
    return page.evaluate(() => {
      const key = Object.keys(localStorage).find((name) => name.endsWith(":feedback"));
      return key ? { key, value: JSON.parse(localStorage.getItem(key)) } : null;
    });
  };
  const prompt = () => page.locator("#prompt-text").inputValue();
  const count = async () => (await page.locator("#answered-count").innerText()).trim();
  return { context, page, problems, open, stored, prompt, count };
}
