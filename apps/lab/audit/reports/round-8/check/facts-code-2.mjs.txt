function lines(file, from, to) {
  const all = readFileSync(join(root, file), "utf8").split("\n");
  console.log(`--- ${file} lines ${from} to ${to} (file has ${all.length} lines)`);
  for (let index = from; index <= to; index += 1) {
    const text = all[index - 1] ?? "";
    console.log(`${index}: ${text.length > 300 ? `${text.slice(0, 300)} [...]` : text}`);
  }
}

function grep(pattern, paths) {
  try {
    return execFileSync("git", ["-C", root, "grep", "-n", "-E", pattern, "--", ...paths], {
      encoding: "utf8",
      maxBuffer: 1 << 26,
    });
  } catch {
    return "(no match)\n";
  }
}
