// Reads the public state of the two repositories with GET requests only.
import { execFileSync } from "node:child_process";

function gh(args) {
  try {
    return execFileSync("gh", args, { encoding: "utf8", maxBuffer: 1 << 26 }).trim();
  } catch (error) {
    return `ERROR ${String(error.stderr ?? error.message).trim().split("\n")[0]}`;
  }
}

function api(path, query) {
  const args = ["api", "--method", "GET", path];
  if (query) {
    args.push("--jq", query);
  }
  return gh(args);
}

for (const repository of ["ariakit/visonaut", "ariakit/ariakit"]) {

  console.log(`SECURITY.md: ${api(`repos/${repository}/contents/SECURITY.md`, ".path")}`);
  const labels = api(`repos/${repository}/labels?per_page=100`, '.[] | "\\(.name)\\t\\(.color)\\t\\(.description)"');

    console.log(labels);
  } else {
    console.log(labels.split("\n").filter((line) => /^(p0|p1|p2|easy|hard)\t/.test(line)).join("\n"));
  }
}

function fileLines(path, numbers) {
  const text = gh(["api", "--method", "GET", `repos/ariakit/ariakit/contents/${path}?ref=main`, "-H", "Accept: application/vnd.github.raw"]);
  const lines = text.split("\n");
  for (const number of numbers) {
    console.log(`${path}:${number}: ${(lines[number - 1] ?? "").trim().slice(0, 260)}`);
  }
}
