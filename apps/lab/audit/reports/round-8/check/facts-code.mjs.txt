function lines(file, from, to) {
  const absolute = join(root, file);
  if (!existsSync(absolute)) {
    console.log(`MISSING ${file}`);
    return;
  }
  const all = readFileSync(absolute, "utf8").split("\n");
  console.log(`--- ${file} lines ${from} to ${to} (file has ${all.length} lines)`);
  for (let index = from; index <= to; index += 1) {
    const text = all[index - 1] ?? "";
    console.log(`${index}: ${text.length > 420 ? `${text.slice(0, 420)} [...]` : text}`);
  }
}

function words(file) {
  const text = readFileSync(join(root, file), "utf8");
  return text.split(/\s+/).filter(Boolean).length;
}

function countFiles(folder) {
  let count = 0;
  for (const entry of readdirSync(join(root, folder), { withFileTypes: true })) {
    if (entry.isDirectory()) {
      count += countFiles(join(folder, entry.name));
    } else {
      count += 1;
    }
  }
  return count;
}

function git(args) {
  return execFileSync("git", ["-C", root, ...args], { encoding: "utf8", maxBuffer: 1 << 28 });
}
