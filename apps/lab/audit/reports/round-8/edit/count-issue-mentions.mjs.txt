// Counts which public issue texts of the plan name each decision of round 6.
// Read only.
import { readdirSync, readFileSync } from "node:fs";

const folder = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/issues/final";
const files = readdirSync(folder).filter(
  (name) => /^\d\d-.*\.md$/.test(name) && !name.includes(".private."),
);
const issuesWithMention = new Set();
for (let number = 1; number <= 12; number += 1) {
  const id = `D-PRE-${String(number).padStart(2, "0")}`;
  const named = [];
  for (const name of files) {
    const text = readFileSync(`${folder}/${name}`, "utf8");
    if (!text.includes(id)) continue;
    named.push(name);
    if (!name.startsWith("00-")) {
      issuesWithMention.add(name.slice(0, 2));
    }
  }
  console.log(`${id}: ${named.length} files: ${named.join(", ")}`);
}
console.log(`issue numbers with a mention: ${issuesWithMention.size}: ${[...issuesWithMention].join(" ")}`);
const privateFiles = readdirSync(folder).filter((name) => name.includes(".private."));
console.log(`private files: ${privateFiles.join(", ")}`);
