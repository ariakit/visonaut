// Says for each user id in an avatar address of the res-05 probe results if it
// is one of the new invented ids of the lab reviewers. It prints no id.
import { readdirSync, readFileSync } from "node:fs";

const folder =
  "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel/apps/lab/audit/reports/round-2/res-05/probe/results";
const map = JSON.parse(
  readFileSync("/Users/diegohaz/.claude/jobs/f65a6229/tmp/round7/people/map.json", "utf8"),
);
const fresh = new Set(map.people.map((person) => String(person.newId)));
const old = new Set(map.people.map((person) => String(person.oldId)));
const found = new Set();
for (const name of readdirSync(folder)) {
  const text = readFileSync(`${folder}/${name}`, "utf8");
  for (const match of text.matchAll(/avatars\.githubusercontent\.com\/u\/(\d+)/g)) {
    found.add(match[1]);
  }
}
let index = 0;
for (const id of found) {
  index += 1;
  const kind = fresh.has(id) ? "a new invented id" : old.has(id) ? "an OLD id" : "not in the map";
  console.log(`avatar id ${index}: ${kind}`);
}
