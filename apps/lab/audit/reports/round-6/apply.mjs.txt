const readJson = (path) => JSON.parse(readFileSync(path, "utf8"));
const writeJson = (path, value) => {
  if (dry) return;
  writeFileSync(path, `${JSON.stringify(value, null, 2)}\n`);
};

for (const [id, entry] of Object.entries(settled)) {
  const decision = decisions.find((item) => item.id === id);
  if (!decision) throw new Error(`unknown decision ${id}`);
  if (decision.status !== "open") throw new Error(`${id} is not open`);
  if (!decision.options.some((option) => option.id === entry.selection)) {
    throw new Error(`${id}: ${entry.selection} is not an option`);
  }
  decision.status = "settled";
  decision.settledAnswer = entry.settledAnswer;
  decision.rationale = entry.rationale;
  record.incorporated[id] = { selection: entry.selection, notes: entry.notes };
  console.log(`${id}: settled as ${entry.selection}`);
}

record.revision = "r8";
record.date = "2026-10-07";
const open = decisions.filter((decision) => decision.status === "open").map((d) => d.id);
console.log(`settled ${decisions.length - open.length}, open ${open.length}: ${open.join(", ")}`);
writeJson(join(content, "record.json"), record);
writeJson(join(content, "decisions.json"), decisions);
console.log(dry ? "dry run: nothing written" : "record and decisions written");
