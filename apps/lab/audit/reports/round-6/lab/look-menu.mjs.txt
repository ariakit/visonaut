// Round 4 lane: the Look menu on a page with cards and on a page with no
// card, at 1440 px and at 400 px.
// Usage: node look-menu.mjs --origin http://127.0.0.1:4372
import { appendFileSync } from "node:fs";
import { args, check, chromium, load, quiet } from "./clib.mjs";

const options = args();
const origin = options.origin ?? "http://127.0.0.1:4372";
const logFile = `${check}/results/X3-look-menu.log`;
const say = (line) => {
  console.log(line);
  appendFileSync(logFile, `${line}\n`);
};
const sentence = "The pictures of the cards follow the theme only.";
const browser = await chromium.launch({ channel: "chrome", headless: true });
say(`# look menu: origin=${origin} chrome=${browser.version()} load ${load()}`);
for (const path of ["/", "/directions", "/directions/ariakit", "/pages/review", "/components/notice", "/feedback"]) {
  for (const width of [1440, 400]) {
    const context = await browser.newContext({ viewport: { width, height: 800 }, deviceScaleFactor: 2 });
    const page = await context.newPage();
    await page.goto(`${origin}${path}`, { waitUntil: "networkidle" });
    await quiet(page, 800);
    await page.getByRole("button", { name: "Look" }).click();
    await quiet(page, 600);
    const dialog = page.getByRole("dialog", { name: "Look" });
    const shows = await dialog.getByText(sentence).count();
    const sideways = await page.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth);
    const name = path === "/" ? "gallery" : path.slice(1).replaceAll("/", "-");
    await page.screenshot({ path: `${check}/shots/X3-look-${name}-${width}.png` });
    say(`${path} ${width}: the sentence shows ${shows} time(s) | sideways ${sideways}`);
    await context.close();
  }
}
await browser.close();
