// Review lane: pictures and checks of the lab pages with cards.
// Usage: node pageshots.mjs --name S1 --forms after:4375,before:4376,dev:4320
//
// For each form, page, width, and theme: one picture of the first viewport
// at the load, then each card is scrolled into view, then one picture of the
// whole page. The checks: console errors, failed requests, a page that
// scrolls sideways, frames, the pictures of the shown and the hidden theme,
// and the layout shift.
import { writeFileSync } from "node:fs";
import { chromium, lane, load, logger, readArgs, save } from "./lib.mjs";

const options = readArgs();
const name = options.name ?? "pageshots";
const forms = String(options.forms).split(",").map((form) => {
  const [label, port] = form.split(":");
  return { label, port: Number(port) };
});
const paths = String(options.paths ?? "/,/directions,/directions/ariakit").split(",");
const widths = [
  { width: 1440, height: 900 },
  { width: 400, height: 800 },
];
const themes = ["dark", "light"];
const say = logger(name);

const observerScript = `(() => {
  window.__shifts = [];
  try {
    new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.hadRecentInput) continue;
        window.__shifts.push({ value: entry.value, time: entry.startTime, sources: (entry.sources ?? []).map((source) => (source.node?.nodeName ?? "?") + "." + String(source.node?.className ?? "").slice(0, 40)) });
      }
    }).observe({ type: "layout-shift", buffered: true });
  } catch {}
})();`;

const readCards = () => {
  const images = [...document.querySelectorAll('img[src^="/cards/"]')];
  const shown = images.filter((image) => image.getClientRects().length > 0);
  const hidden = images.filter((image) => image.getClientRects().length === 0);
  const loaded = (image) => image.complete && image.naturalWidth > 0;
  const root = document.documentElement;
  const scale = shown.map((image) => {
    const rect = image.getBoundingClientRect();
    return {
      src: image.getAttribute("src"),
      box: [Math.round(rect.width), Math.round(rect.height)],
      natural: [image.naturalWidth, image.naturalHeight],
      // The ratio of the box against the ratio of the file: 1 is no distortion.
      ratio: image.naturalWidth
        ? rect.width / rect.height / (image.naturalWidth / image.naturalHeight)
        : null,
      pixelsForEachCssPixel: image.naturalWidth ? image.naturalWidth / rect.width : null,
    };
  });
  return {
    iframes: document.querySelectorAll("iframe").length,
    images: images.length,
    shown: shown.length,
    shownLoaded: shown.filter(loaded).length,
    shownTheme: [...new Set(shown.map((image) => image.getAttribute("src").match(/\.(dark|light)\.png$/)?.[1]))],
    hiddenLoaded: hidden.filter(loaded).length,
    sideways: root.scrollWidth > root.clientWidth,
    scrollWidth: root.scrollWidth,
    clientWidth: root.clientWidth,
    theme: root.dataset.theme,
    scale,
  };
};

async function once(browser, form, route, size, theme) {
  const context = await browser.newContext({ viewport: size });
  await context.addInitScript(observerScript);
  const page = await context.newPage();
  const problems = [];
  const failed = [];
  const notGet = [];
  const pictureRequests = [];
  page.on("pageerror", (error) => problems.push(String(error).slice(0, 200)));
  page.on("console", (message) => {
    if (message.type() === "error") problems.push(message.text().slice(0, 200));
  });
  page.on("requestfailed", (request) => failed.push(request.url().slice(0, 120)));
  page.on("response", (response) => {
    if (response.status() >= 400) failed.push(`${response.status()} ${response.url().slice(0, 120)}`);
  });
  page.on("request", (request) => {
    if (request.method() !== "GET") notGet.push(request.method());
    const pathname = new URL(request.url()).pathname;
    if (pathname.startsWith("/cards/")) pictureRequests.push(pathname);
  });
  await page.goto(`http://127.0.0.1:${form.port}${route}?theme=${theme}`, { waitUntil: "networkidle" });
  await page.waitForTimeout(1200);
  const slug = `${name}-${form.label}-${route === "/" ? "gallery" : route.slice(1).replaceAll("/", "-")}-${size.width}-${theme}`;
  const atLoad = await page.evaluate(readCards);
  const requestsAtLoad = pictureRequests.length;
  writeFileSync(`${lane}/shots/${slug}-first.png`, await page.screenshot());
  // Each card comes into view one time, so each lazy picture loads.
  // The box of a card is the parent of its pictures or of its frame. A
  // picture of the hidden theme has no box, so the script scrolls the parent.
  const count = await page.evaluate(
    () => new Set([...document.querySelectorAll('img[src^="/cards/"], iframe')].map((item) => item.parentElement)).size,
  );
  for (let i = 0; i < count; i += 1) {
    await page.evaluate((index) => {
      const boxes = [...new Set([...document.querySelectorAll('img[src^="/cards/"], iframe')].map((item) => item.parentElement))];
      boxes[index]?.scrollIntoView({ block: "center" });
    }, i);
    await page.waitForTimeout(150);
  }
  await page.waitForLoadState("networkidle");
  await page.waitForTimeout(form.label === "before" ? 4000 : 800);
  const afterScroll = await page.evaluate(readCards);
  const shifts = await page.evaluate(() => window.__shifts ?? []);
  await page.evaluate(() => {
    for (const element of document.querySelectorAll("*")) {
      if (element.scrollTop > 0) element.scrollTop = 0;
    }
    window.scrollTo(0, 0);
  });
  await page.waitForTimeout(300);
  writeFileSync(`${lane}/shots/${slug}-full.png`, await page.screenshot({ fullPage: true }));
  await context.close();
  return {
    slug,
    atLoad,
    requestsAtLoad,
    afterScroll,
    pictureRequests,
    shift: shifts.reduce((sum, entry) => sum + entry.value, 0),
    shifts,
    problems,
    failed,
    notGet,
  };
}

const browser = await chromium.launch({ channel: "chrome", headless: true });
say(`# ${name}: forms=${options.forms} chrome=${browser.version()} load ${load()}`);
const results = [];
let bad = 0;
for (const form of forms) {
  for (const route of paths) {
    for (const size of widths) {
      for (const theme of themes) {
        const result = await once(browser, form, route, size, theme);
        results.push({ form: form.label, route, width: size.width, theme, ...result });
        const { atLoad, afterScroll } = result;
        const issues = [];
        if (result.problems.length) issues.push(`console ${JSON.stringify(result.problems)}`);
        if (result.failed.length) issues.push(`failed ${JSON.stringify(result.failed)}`);
        if (result.notGet.length) issues.push(`not GET ${result.notGet.join(",")}`);
        if (afterScroll.sideways) issues.push(`scrolls sideways ${afterScroll.scrollWidth} > ${afterScroll.clientWidth}`);
        if (form.label !== "before") {
          if (afterScroll.iframes) issues.push(`${afterScroll.iframes} frames`);
          if (afterScroll.shownLoaded !== afterScroll.shown) issues.push(`shown ${afterScroll.shown}, loaded ${afterScroll.shownLoaded}`);
          if (afterScroll.hiddenLoaded) issues.push(`${afterScroll.hiddenLoaded} pictures of the hidden theme loaded`);
          if (afterScroll.shownTheme.join() !== theme) issues.push(`shown theme ${afterScroll.shownTheme.join()}`);
          if (afterScroll.theme !== theme) issues.push(`page theme ${afterScroll.theme}`);
          const distorted = afterScroll.scale.filter((item) => item.ratio && Math.abs(item.ratio - 1) > 0.01);
          if (distorted.length) issues.push(`distorted ${JSON.stringify(distorted)}`);
          const blurred = afterScroll.scale.filter((item) => item.pixelsForEachCssPixel && item.pixelsForEachCssPixel < 2);
          if (blurred.length) issues.push(`below 2 picture pixels for each CSS pixel: ${JSON.stringify(blurred.map((item) => [item.src, item.box, item.natural]))}`);
        }
        if (issues.length) bad += 1;
        const densities = afterScroll.scale.map((item) => item.pixelsForEachCssPixel).filter(Boolean);
        say(
          `${result.slug}: frames ${afterScroll.iframes} | pictures in the DOM ${afterScroll.images}, shown ${afterScroll.shown} | loaded at the load ${atLoad.shownLoaded} (requests ${result.requestsAtLoad}), after the scroll ${afterScroll.shownLoaded} (requests ${result.pictureRequests.length}), hidden theme loaded ${afterScroll.hiddenLoaded} | picture pixels for each CSS pixel ${densities.length ? `${Math.min(...densities).toFixed(2)} to ${Math.max(...densities).toFixed(2)}` : "-"} | shift ${result.shift.toFixed(4)}${result.shifts.length ? ` ${JSON.stringify(result.shifts.map((entry) => [Math.round(entry.time), entry.value.toFixed(4), entry.sources.slice(0, 2)]))}` : ""} | ${issues.length ? `PROBLEMS: ${issues.join("; ")}` : "no problem"} | load ${load()}`,
        );
      }
    }
  }
}
await browser.close();
say(`\n${bad} of ${results.length} page states with a problem`);
save(name, results);
