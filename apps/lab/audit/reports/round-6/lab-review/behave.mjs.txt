// Review lane: how a card behaves. GET requests only.
// Usage: node behave.mjs --name B1 --port 4375
//
// 1. Theme button: which pictures show and which are requested.
// 2. A missing picture (HTTP 404 from the script, and a failed request).
// 3. Keyboard and names: Tab to a card, its name, its focus ring, Enter.
// 4. The explorers still have a live frame.
// 5. The Look menu sentence, and a changed look with a card.
import { writeFileSync } from "node:fs";
import { chromium, lane, load, logger, readArgs } from "./lib.mjs";

const options = readArgs();
const name = options.name ?? "behave";
const port = Number(options.port ?? 4375);
const origin = `http://127.0.0.1:${port}`;
const say = logger(name);
const browser = await chromium.launch({ channel: "chrome", headless: true });
say(`# ${name}: ${origin} chrome=${browser.version()} load ${load()}`);

const readCards = () => {
  const images = [...document.querySelectorAll('img[src^="/cards/"]')];
  const shown = images.filter((image) => image.getClientRects().length > 0);
  const loaded = (image) => image.complete && image.naturalWidth > 0;
  return {
    theme: document.documentElement.dataset.theme,
    inDom: images.length,
    shown: shown.length,
    shownLoaded: shown.filter(loaded).length,
    shownThemes: [...new Set(shown.map((image) => image.src.match(/\.(dark|light)\.png$/)?.[1]))],
    hiddenLoaded: images.filter((image) => !shown.includes(image) && loaded(image)).length,
  };
};

async function newPage(size = { width: 1440, height: 900 }) {
  const context = await browser.newContext({ viewport: size });
  const page = await context.newPage();
  const problems = [];
  const pictures = [];
  const notGet = [];
  page.on("pageerror", (error) => problems.push(String(error).slice(0, 200)));
  page.on("console", (message) => {
    if (message.type() === "error") problems.push(message.text().slice(0, 200));
  });
  page.on("request", (request) => {
    if (request.method() !== "GET") notGet.push(request.method());
    const pathname = new URL(request.url()).pathname;
    if (pathname.startsWith("/cards/")) pictures.push(pathname);
  });
  return { context, page, problems, pictures, notGet };
}

async function scrollCards(page) {
  // A picture of the hidden theme has no box, so the script scrolls the box
  // of the card, which is the parent of its two pictures.
  const count = await page.evaluate(
    () => new Set([...document.querySelectorAll('img[src^="/cards/"]')].map((item) => item.parentElement)).size,
  );
  for (let i = 0; i < count; i += 1) {
    await page.evaluate((index) => {
      const boxes = [...new Set([...document.querySelectorAll('img[src^="/cards/"]')].map((item) => item.parentElement))];
      boxes[index]?.scrollIntoView({ block: "center" });
    }, i);
    await page.waitForTimeout(120);
  }
  await page.waitForLoadState("networkidle");
  await page.waitForTimeout(500);
}

// 1. Theme button.
for (const route of ["/", "/directions"]) {
  const { context, page, problems, pictures } = await newPage();
  await page.goto(origin + route, { waitUntil: "networkidle" });
  await page.waitForTimeout(800);
  say(`theme ${route} at the load: ${JSON.stringify(await page.evaluate(readCards))} | picture requests ${pictures.length}`);
  await scrollCards(page);
  say(`theme ${route} after the scroll: ${JSON.stringify(await page.evaluate(readCards))} | picture requests ${pictures.length}`);
  for (let press = 1; press <= 3; press += 1) {
    const before = pictures.length;
    await page.getByRole("button", { name: /^Switch to (light|dark) theme$/ }).click();
    await page.waitForLoadState("networkidle");
    await page.waitForTimeout(700);
    await scrollCards(page);
    const state = await page.evaluate(readCards);
    say(`theme ${route} press ${press}: ${JSON.stringify(state)} | new picture requests ${pictures.length - before}`);
    if (press === 1) {
      await page.evaluate(() => {
        for (const element of document.querySelectorAll("*")) {
          if (element.scrollTop > 0) element.scrollTop = 0;
        }
      });
      writeFileSync(`${lane}/shots/${name}-theme-${route === "/" ? "gallery" : "directions"}-after-press-1.png`, await page.screenshot());
    }
  }
  if (problems.length) say(`theme ${route} problems: ${JSON.stringify(problems)}`);
  // A reload keeps the saved theme (light after three presses).
  await page.reload({ waitUntil: "networkidle" });
  await page.waitForTimeout(800);
  say(`theme ${route} after a reload with the saved theme: ${JSON.stringify(await page.evaluate(readCards))}`);
  await context.close();
}

// 2. A missing picture.
for (const kind of ["404", "failed", "html-200"]) {
  const { context, page, problems } = await newPage();
  await page.route("**/cards/page/inbox/**", (route) => {
    if (kind === "404") return route.fulfill({ status: 404, contentType: "text/plain", body: "Not found" });
    if (kind === "html-200") return route.fulfill({ status: 200, contentType: "text/html", body: "<!doctype html><title>x</title>" });
    return route.abort();
  });
  await page.goto(`${origin}/`, { waitUntil: "networkidle" });
  await page.waitForTimeout(1000);
  const state = await page.evaluate(() => {
    const cards = [...document.querySelectorAll("main a[aria-label]")].map((link) => link.parentElement);
    return cards.map((card) => {
      const box = card.firstElementChild;
      const rect = box.getBoundingClientRect();
      const images = [...box.querySelectorAll("img")];
      return {
        name: card.querySelector("a[aria-label]").getAttribute("aria-label"),
        box: [Math.round(rect.width), Math.round(rect.height)],
        images: images.length,
        shownImages: images.filter((image) => image.getClientRects().length > 0).length,
      };
    });
  });
  say(`missing (${kind}): ${JSON.stringify(state)}`);
  writeFileSync(`${lane}/shots/${name}-missing-${kind}.png`, await page.screenshot());
  // The picture of the other theme is missing too. The card must keep its box.
  await page.getByRole("button", { name: /^Switch to light theme$/ }).click();
  await page.waitForTimeout(1200);
  const light = await page.evaluate(() => {
    const link = document.querySelector('main a[aria-label="Queue"], main a[aria-label="Inbox"]') ?? document.querySelector("main a[aria-label]");
    const box = link.parentElement.firstElementChild;
    const rect = box.getBoundingClientRect();
    return { name: link.getAttribute("aria-label"), box: [Math.round(rect.width), Math.round(rect.height)], images: box.querySelectorAll("img").length };
  });
  say(`missing (${kind}), light theme: ${JSON.stringify(light)} | console ${JSON.stringify(problems)}`);
  await context.close();
}

// 3. Keyboard and names.
{
  const { context, page } = await newPage();
  await page.goto(`${origin}/`, { waitUntil: "networkidle" });
  await page.waitForTimeout(800);
  const names = await page.evaluate(() =>
    [...document.querySelectorAll("main a")].map((link) => ({
      label: link.getAttribute("aria-label"),
      text: link.textContent.trim().slice(0, 40),
      href: link.getAttribute("href"),
    })),
  );
  say(`links of the gallery: ${JSON.stringify(names)}`);
  const alts = await page.evaluate(() =>
    [...document.querySelectorAll("main img")].map((image) => image.getAttribute("alt")),
  );
  say(`alt of each picture: ${JSON.stringify([...new Set(alts)])} (${alts.length} pictures)`);
  const tree = await page.locator("main").ariaSnapshot();
  writeFileSync(`${lane}/results/${name}-aria-gallery.txt`, tree);
  say(`aria tree of the gallery has "img": ${/\bimg\b/.test(tree)} | links: ${(tree.match(/- link /g) ?? []).length}`);
  await page.getByRole("searchbox", { name: "Filter surfaces" }).focus();
  await page.keyboard.press("Tab");
  const focused = await page.evaluate(() => {
    const element = document.activeElement;
    const style = getComputedStyle(element);
    return {
      tag: element.tagName,
      label: element.getAttribute("aria-label"),
      focusVisible: element.matches(":focus-visible"),
      outline: `${style.outlineStyle} ${style.outlineWidth} ${style.outlineColor} offset ${style.outlineOffset}`,
    };
  });
  say(`after Tab from the filter: ${JSON.stringify(focused)}`);
  writeFileSync(`${lane}/shots/${name}-focus-card.png`, await page.screenshot({ clip: { x: 0, y: 0, width: 1440, height: 520 } }));
  await page.keyboard.press("Enter");
  await page.waitForURL(/\/pages\//, { timeout: 15_000 });
  await page.waitForLoadState("networkidle");
  await page.waitForTimeout(1500);
  say(`Enter opens ${new URL(page.url()).pathname} | frames in the page explorer: ${await page.locator("iframe").count()} | card pictures: ${await page.locator('img[src^="/cards/"]').count()}`);
  await context.close();
}

// 4. The explorers.
for (const route of ["/pages/review", "/pages/review?compare=1", "/components/status-mark", "/primitives", "/feedback"]) {
  const { context, page, problems } = await newPage();
  await page.goto(origin + route, { waitUntil: "networkidle" });
  await page.waitForTimeout(1500);
  const state = await page.evaluate(() => ({
    iframes: [...document.querySelectorAll("iframe")].map((frame) => frame.getAttribute("src")?.slice(0, 70)),
    cardPictures: document.querySelectorAll('img[src^="/cards/"]').length,
  }));
  say(`${route}: ${JSON.stringify(state)}${problems.length ? ` | console ${JSON.stringify(problems)}` : ""}`);
  await context.close();
}

// 5. The Look menu.
for (const size of [{ width: 1440, height: 900 }, { width: 400, height: 800 }]) {
  for (const route of ["/", "/directions", "/directions/ariakit", "/pages/inbox", "/components/notice", "/feedback", "/primitives"]) {
    const { context, page } = await newPage(size);
    await page.goto(origin + route, { waitUntil: "networkidle" });
    await page.waitForTimeout(600);
    await page.getByRole("button", { name: "Look" }).click();
    const dialog = page.getByRole("dialog", { name: "Look" });
    await dialog.waitFor();
    await page.waitForTimeout(400);
    const text = await dialog.innerText();
    const has = text.includes("The pictures of the cards follow the theme only.");
    const box = await dialog.boundingBox();
    const sideways = await page.evaluate(() => document.documentElement.scrollWidth > document.documentElement.clientWidth);
    say(`look menu ${route} at ${size.width}: sentence ${has} | menu box ${Math.round(box.x)},${Math.round(box.y)} ${Math.round(box.width)}x${Math.round(box.height)} | page scrolls sideways ${sideways}`);
    if (route === "/" || route === "/pages/inbox") {
      writeFileSync(`${lane}/shots/${name}-look-${route === "/" ? "gallery" : "pages-inbox"}-${size.width}.png`, await page.screenshot());
    }
    await context.close();
  }
}

await browser.close();
say(`done, load ${load()}`);
