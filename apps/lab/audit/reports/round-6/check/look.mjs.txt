// Pictures of the places that revision r8 changed, at 1440 px and at 400 px.
// It reads the standalone output from the lab server of port 4320 with GET
// requests only. Usage: node look.mjs
import { mkdirSync } from "node:fs";
import { createRequire } from "node:module";

const root = "/Users/diegohaz/Developer/visonaut/.claude/worktrees/serialized-dazzling-pixel";
const require = createRequire(`${root}/package.json`);
const { chromium } = require("@playwright/test");
const out = "/Users/diegohaz/.claude/jobs/f65a6229/tmp/round6/check/shots";
mkdirSync(out, { recursive: true });

// A viewport picture after a jump to the target.
const views = ["top", "summary-start", "your-answers", "your-answers-holds", "your-answers-settled", "ui-system-speed", "ui-system-speed-list", "ui-system-speed-list-built", "ui-system-speed-test", "ui-system-speed-gallery", "ui-system-speed-gallery-built", "design-lab", "design-lab-pictures", "record", "feedback"];
// A picture of one whole element.
const elements = ["decision-D-PERF-01", "decision-D-PERF-02", "decision-D-PERF-03"];

const browser = await chromium.launch({ channel: "chrome", headless: true });
for (const viewport of [
  { width: 1440, height: 900 },
  { width: 400, height: 800 },
]) {
  const context = await browser.newContext({ viewport, colorScheme: "dark" });
  await context.route("**/*", (route) => {
    if (route.request().method() === "GET") {
      return route.continue();
    }
    return route.abort();
  });
  const page = await context.newPage();
  const errors = [];
  page.on("console", (message) => {
    if (message.type() === "error") errors.push(message.text());
  });
  await page.goto("http://127.0.0.1:4320/audit/index.html", { waitUntil: "load" });
  await page.locator("#doc[data-ready]").waitFor({ timeout: 15000 });
  await page.waitForTimeout(1200);
  for (const id of views) {
    await page.evaluate((target) => {
      location.hash = `#${target}`;
    }, id);
    await page.waitForTimeout(500);
    await page.screenshot({ path: `${out}/${viewport.width}-${id}.png` });
  }
  for (const id of elements) {
    await page.locator(`[id="${id}"]`).screenshot({ path: `${out}/${viewport.width}-${id}.png` });
  }
  // The rows of the three decisions in the decision record.
  const item = page.locator("#decision-record-list .record-item", { hasText: "D-PERF-03" }).first();
  await item.scrollIntoViewIfNeeded();
  await item.screenshot({ path: `${out}/${viewport.width}-record-D-PERF-03.png` });
  // Step 8 of the order demo.
  const demo = page.locator("#demo-answers-order");
  await demo.scrollIntoViewIfNeeded();
  await demo.screenshot({ path: `${out}/${viewport.width}-demo-order.png` });
  const facts = await page.evaluate(() => {
    const wide = [];
    for (const element of document.querySelectorAll("#main *")) {
      const box = element.getBoundingClientRect();
      if (box.width === 0) continue;
      if (box.right <= document.documentElement.clientWidth + 0.5) continue;
      let scroller = element.parentElement;
      let contained = false;
      while (scroller && scroller.id !== "main") {
        const overflow = getComputedStyle(scroller).overflowX;
        if (overflow === "auto" || overflow === "scroll" || overflow === "hidden") contained = true;
        scroller = scroller.parentElement;
      }
      if (!contained) wide.push(`${element.tagName}.${element.className}`.slice(0, 60));
    }
    return {
      sideways: document.documentElement.scrollWidth - document.documentElement.clientWidth,
      wide: wide.slice(0, 5),
      bar: document.getElementById("bar").innerText.replace(/\s+/g, " "),
      theme: getComputedStyle(document.documentElement).colorScheme,
    };
  });
  console.log(`${viewport.width}: ${JSON.stringify(facts)} | console errors ${errors.length} ${errors.slice(0, 2).join(" | ")}`);
  await context.close();
}
await browser.close();
